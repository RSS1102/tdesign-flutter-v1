156  @override


157  void paint(Canvas canvas, Size size) {


158    final paint = Paint();


159    final tickCount = _kAlphaValues.length;


160


161    canvas.save();


162    canvas.translate(size.width / 2.0, size.height / 2.0);


163


164    final activeTick = (tickCount * position.value).floor();


165


166    for (var i = 0; i < tickCount * progress; ++i) {


167      final t = (i - activeTick) % tickCount;


168      paint.color = activeColor


169          .withAlpha(progress < 1 ? _partiallyRevealedAlpha : _kAlphaValues[t]);


170      canvas.drawRRect(tickFundamentalRRect, paint);


171      canvas.rotate(_kTwoPI / tickCount);


172    }


173


174    canvas.restore();


175  }
