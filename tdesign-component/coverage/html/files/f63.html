33/// 各列选项步进，未配置的列步进为 1。


34@immutable


35class DateTimePickerSteps {

46  /// 年列步进。


47  final int? year;

49  /// 月列步进。

64  @internal


65  int forColumn(DateTimeColumn column) {


66    final step = switch (column) {


67      DateTimeColumn.year => year,


68      DateTimeColumn.month => month,


69      DateTimeColumn.day => day,


70      DateTimeColumn.hour => hour,


71      DateTimeColumn.minute => minute,


72      DateTimeColumn.second => second,


73    };


74    return step == null


75        ? 1


76        : step < 1


77            ? 1


78            : step;


79  }

88          hour == other.hour &&
