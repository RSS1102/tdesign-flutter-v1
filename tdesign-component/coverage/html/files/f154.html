429  Widget _buildLabelWidget(Color labelColor) {


430    late double iconSize;


431    late double fontSize;


432    late FontWeight fontWeight;


433


434    switch (widget.type) {


435      case TProgressVariant.linear:


436        if (widget.progressLabelPosition != TProgressLabelPosition.inside) {


437          fontSize = widget.strokeWidth > 14 ? widget.strokeWidth : 14;


438          iconSize = widget.strokeWidth > 20 ? widget.strokeWidth : 20;


439        } else {


440          fontSize = widget.strokeWidth * 0.6;


441          iconSize = widget.strokeWidth;


442        }


443        fontWeight = FontWeight.normal;


444        break;


445      case TProgressVariant.circular:


446        iconSize = widget.circleRadius * 0.4;


447        fontSize = widget.circleRadius * 0.15;


448        fontWeight = FontWeight.bold;


449        break;


450      case TProgressVariant.micro:


451        iconSize = widget.circleRadius * 0.5;


452        fontSize = widget.circleRadius * 0.2;


453        fontWeight = FontWeight.normal;


454        break;


455      case TProgressVariant.button:


456        iconSize = widget.strokeWidth * 0.3;


457        fontSize = widget.strokeWidth * 0.3;


458        fontWeight = FontWeight.normal;


459        break;


460    }


461


462    return IconTheme(


463      data: IconThemeData(color: _effectiveColor, size: iconSize),


464      child: DefaultTextStyle(


465        style: TextStyle(


466          color: labelColor,


467          fontSize: fontSize,


468          fontWeight: fontWeight,


469        ),


470        child: _effectiveLabel,


471      ),


472    );


473  }
