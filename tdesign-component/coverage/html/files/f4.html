4/// center 由 [PopupCenterUnderClose] 接管面板外下方关闭区。

12  final TPopupOptions options;

15  @override


16  Widget build(BuildContext context) {

18    final radius = options.radius ?? theme.radiusExtraLarge;


19    final backgroundColor = options.backgroundColor ?? theme.bgColorContainer;

21


22    if (options.placement == TPopupPlacement.center) {

29  Widget _buildCenter(BuildContext context, double radius, Color background) {

32        color: background,


33        borderRadius: BorderRadius.circular(radius),

35      clipBehavior: Clip.antiAlias,


36      child: options.child,


37    );


38


39    if (options.closeBuilder != null) {


40      return PopupCenterUnderClose(


41        options: options,


42        content: panel,


43        onCloseWithTrigger: onCloseWithTrigger,


44      );


45    }


46    return SizedBox(


47      width: options.width,


48      height: options.height,


49      child: panel,


50    );


51  }

53  Widget _buildEdge(


54    BuildContext context,


55    BorderRadius? borderRadius,


56    Color background,


57  ) {


58    final useExpanded = options.placement == TPopupPlacement.left ||


59        options.placement == TPopupPlacement.right ||


60        options.height != null;


61

63        ? Column(


64            mainAxisSize: useExpanded ? MainAxisSize.max : MainAxisSize.min,


65            crossAxisAlignment: CrossAxisAlignment.stretch,


66            children: [


67              PopupHeader(


68                options: options,


69                onCloseWithTrigger: onCloseWithTrigger,


70              ),


71              if (useExpanded) Expanded(child: options.child) else options.child,


72            ],


73          )


74        : (useExpanded

76                mainAxisSize: MainAxisSize.max,


77                crossAxisAlignment: CrossAxisAlignment.stretch,


78                children: [Expanded(child: options.child)],

81


82    return Container(

84        color: background,

92  BorderRadius? _borderRadius(TPopupPlacement placement, double radius) {


93    switch (placement) {


94      case TPopupPlacement.top:


95        return BorderRadius.vertical(bottom: Radius.circular(radius));


96      case TPopupPlacement.bottom:


97        return BorderRadius.vertical(top: Radius.circular(radius));


98      case TPopupPlacement.left:


99        return BorderRadius.horizontal(right: Radius.circular(radius));


100      case TPopupPlacement.right:


101        return BorderRadius.horizontal(left: Radius.circular(radius));


102      case TPopupPlacement.center:


103        return BorderRadius.circular(radius);


104    }


105  }
