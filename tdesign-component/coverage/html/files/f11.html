604  String? _validatePlacementParams() {


605    switch (placement) {


606      case TPopupPlacement.top:


607        if (width != null) {


608          return 'width is not valid for placement=top; use height + inset.';


609        }


610        if (inset != null && inset is! TPopupTopInset) {


611          return 'inset must be TPopupTopInset for placement=top.';


612        }


613        break;


614      case TPopupPlacement.bottom:


615        if (width != null) {


616          return 'width is not valid for placement=bottom; use height + inset.';


617        }


618        if (inset != null && inset is! TPopupBottomInset) {


619          return 'inset must be TPopupBottomInset for placement=bottom.';


620        }


621        break;


622      case TPopupPlacement.left:


623        if (height != null) {


624          return 'height is not valid for placement=left; use width + inset.';


625        }


626        if (inset != null && inset is! TPopupLeftInset) {


627          return 'inset must be TPopupLeftInset for placement=left.';


628        }


629        break;


630      case TPopupPlacement.right:


631        if (height != null) {


632          return 'height is not valid for placement=right; use width + inset.';


633        }


634        if (inset != null && inset is! TPopupRightInset) {


635          return 'inset must be TPopupRightInset for placement=right.';


636        }


637        break;


638      case TPopupPlacement.center:


639        if (inset != null) {


640          return 'inset is not valid for placement=center.';


641        }


642        break;


643    }


644    final hasBottomHeaderCustom = !_isPopupDefaultHeader(headerBuilder) ||


645        titleWidget != null ||


646        !_isPopupDefaultCancel(cancelBuilder) ||


647        !_isPopupDefaultConfirm(confirmBuilder);


648    if (placement != TPopupPlacement.bottom && hasBottomHeaderCustom) {


649      return 'header/titleWidget/cancel/confirmBuilder only apply to '


650          'placement=bottom (got placement=$placement).';


651    }


652    if (placement != TPopupPlacement.center &&


653        !_isPopupDefaultClose(closeBuilder)) {


654▶       return 'closeBuilder only applies to placement=center '


655▶           '(got placement=$placement).';


656    }


657    if (showOverlay && !modal) {


658      return 'showOverlay=true requires modal=true.';


659    }


660    if (!showOverlay && _closeOnOverlayClick == true) {


661      return 'closeOnOverlayClick=true requires showOverlay=true.';


662    }


663    return null;


664  }
