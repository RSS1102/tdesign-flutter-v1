18/// 悬浮操作按钮组件


19///


20/// T2 组合模式：定位层（右下角悬浮 + 可选拖拽/吸附/边界）+ 动作层（默认内嵌 TButton）


21///


22/// 示例：


23/// ```dart


24/// // 纯图标悬浮按钮


25/// Stack(fit: StackFit.expand, children: [


26///   // 页面内容 ...


27///   const TFab(),


28/// ])


29///


30/// // 图标 + 文字 + 自定义配色


31/// TFab(


32///   text: '发布',


33///   buttonProps: TButtonProps(colorScheme: TButtonColorScheme.danger),


34/// )


35///


36/// // 可拖拽悬浮按钮


37/// TFab(


38///   draggable: true,


39///   magnet: true,


40/// )


41/// ```


42class TFab extends StatelessWidget {

44    super.key,

46    this.icon,


47    this.child,

62  /// 图标 + 文字形态；非空时内嵌 TButton 为 round 形状


63  final String text;


64


65  /// 图标；未传时默认 [Icons.add]


66  final Widget? icon;


67


68  /// 自定义内容；有则替代默认内嵌 TButton，忽略 [buttonProps]


69  final Widget? child;

71  /// 内嵌 TButton 的部分配置透传


72  final TButtonProps? buttonProps;


73


74  /// 点击回调，null 时禁用


75  final VoidCallback? onPressed;


76


77  /// 纯图标 Fab 的 tooltip 提示


78  final String? tooltip;


79


80  /// 读屏标签


81  final String? semanticLabel;


82


83  /// 距屏幕右侧偏移（默认 16）


84  final double? right;

86  /// 距屏幕底部偏移（默认 32）


87  final double? bottom;

89  /// 拖拽轴向：true/[TFabDragAxis.all] 全向，[TFabDragAxis.vertical]/[TFabDragAxis.horizontal] 单轴

107  @override


108  Widget build(BuildContext context) {


109    // ---- resolve Theme ----


110    final theme = Theme.of(context).extension<TFabThemeData>();


111    final safePadding = MediaQuery.of(context).padding;


112


113    // ---- resolveLayout ----


114    final layout = TFabResolve.resolveLayout(


115      right: right,


116      bottom: bottom,


117      draggable: draggable,


118      magnet: magnet,


119      xBounds: xBounds,


120      yBounds: yBounds,


121      themeDefaultRight: theme?.defaultRight,


122      themeDefaultBottom: theme?.defaultBottom,


123      themeDefaultXBounds: theme?.defaultXBounds,


124      themeDefaultYBounds: theme?.defaultYBounds,


125      safePadding: safePadding,


126    );


127


128    final dragTapSlop = theme?.dragTapSlop ?? 18;


129    final magnetDuration =


130        theme?.magnetAnimationDuration ?? const Duration(milliseconds: 200);


131


132    // ---- 构建动作层 ----


133    final isChildMode = child != null;


134    Widget actionChild;


135


136    if (isChildMode) {


137      actionChild = child!;


138    } else {


139      actionChild = TFabResolve.resolveButton(


140        text: text,


141        icon: icon,


142        buttonProps: buttonProps,


143        onPressed: onPressed,


144        context: context,


145      );


146    }


147


148    // 禁用时包 IgnorePointer（仅 child 模式，TButton 内部已处理禁用）


149    if (isChildMode && onPressed == null) {


150      actionChild = IgnorePointer(child: actionChild);


151    }


152


153    // Tooltip / Semantics


154    if (tooltip != null && tooltip!.isNotEmpty) {


155      actionChild = Tooltip(message: tooltip!, child: actionChild);


156    }


157    if (semanticLabel != null && semanticLabel!.isNotEmpty) {


158      actionChild = Semantics(label: semanticLabel!, child: actionChild);


159    }


160


161    // ---- 定位层 ----


162    return buildFabPositioned(


163      layout: layout,


164      child: actionChild,


165      dragTapSlop: dragTapSlop,


166      onPressed: isChildMode ? onPressed : null,


167      onDragStart: onDragStart,


168      onDragEnd: onDragEnd,


169      magnetAnimationDuration: magnetDuration,


170    );


171  }
