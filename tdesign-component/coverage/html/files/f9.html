39    if (options.overlayOpacity != null) {

195  void _onAnimationStatus(AnimationStatus status) {


196    if (status == AnimationStatus.completed && !_openedFired) {


197      _openedFired = true;


198      options.onOpened?.call();


199    }


200    if (status == AnimationStatus.dismissed && !_closedFired) {


201      _closedFired = true;


202      options.onClosed?.call();


203    }


204  }

206  void _attachAnimationListener() {


207    if (_animationListenerAttached) {


208      return;


209    }


210    final anim = animation;


211    if (anim == null) {


212      return;


213    }


214    _animationListenerAttached = true;


215    anim.addStatusListener(_onAnimationStatus);


216    if (anim.status == AnimationStatus.completed) {


217      _onAnimationStatus(AnimationStatus.completed);


218    }


219  }
