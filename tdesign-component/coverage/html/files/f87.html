276  /// 打开菜单


277  Future<void> _openMenu(int index) async {


278    /// 如果已经打开了，则关闭


279    if (_isOpened.contains(true)) {


280▶       await Navigator.maybePop(context);


281    }


282    _dropdownPopup ??= TDropdownPopup(


283      child: _items![index],


284      parentContext: context,


285      handleClose: _closeMenu,


286      direction: widget.direction,


287      showOverlay: widget.showOverlay,


288      closeOnClickOverlay: widget.closeOnClickOverlay,


289      duration: Duration(milliseconds: (widget.duration ?? 200).toInt()),


290    );


291    unawaited(_dropdownPopup!.add(_items![index]).then((value) {


292      widget.onMenuOpened?.call(index);


293    }));


294


295    _isOpened = List.filled(_items?.length ?? 0, false);


296    _isOpened[index] = true;


297    setState(() {});


298    _iconControllers?.asMap().forEach((key, value) {


299      if (value.status == AnimationStatus.completed) {


300▶         value.reverse();


301      } else if (key == index) {


302        value.forward();


303      }


304    });


305  }

308  Future<void> _closeMenu() async {


309    var index = _isOpened.indexOf(true);


310    if (index < 0) {


311      return;


312    }


313    _isOpened = List.filled(_items?.length ?? 0, false);


314    setState(() {});


315    _iconControllers?.forEach((value) {


316      if (value.status == AnimationStatus.completed) {


317        value.reverse();


318      }


319    });


320    await _dropdownPopup?.remove();


321    if (index >= 0 && widget.onMenuClosed != null) {


322▶       widget.onMenuClosed!(index);


323    }


324  }
