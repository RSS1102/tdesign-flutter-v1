240    final matches = _timeReg.allMatches(format);


241    final timeMap = _getTimeMap(matches.map((e) => e.group(0) ?? '').toList());


242    return matches

251        })

300  Map<String, String> _getTimeMap(List<String> timeType) {


301    var duration = Duration(milliseconds: _time);


302    final map = <String, String>{};


303    final dayKey = timeType.find((item) => item.startsWith('D'));


304    final hourKey = timeType.find((item) => item.startsWith('H'));


305    final minuteKey = timeType.find((item) => item.startsWith('m'));


306    final secondKey = timeType.find((item) => item.startsWith('s'));


307    final millisecondKey = timeType.find((item) => item.startsWith('S'));


308    if (dayKey != null) {


309      final length = dayKey.length;


310      map[dayKey] = _toDigits(duration.inDays, length);


311      duration = duration - Duration(days: duration.inDays);


312    }


313    if (hourKey != null) {


314      final length = hourKey.length;


315      final upNum = length > 2 ? pow(10, length).toInt() : 24;


316      final time = duration.inHours.remainder(upNum);


317      map[hourKey] = _toDigits(time, length);


318      duration = duration - Duration(hours: time);


319    }


320    if (minuteKey != null) {


321      final length = minuteKey.length;


322      final upNum = length > 2 ? pow(10, length).toInt() : 60;


323      final time = duration.inMinutes.remainder(upNum);


324      map[minuteKey] = _toDigits(time, length);


325      duration = duration - Duration(minutes: time);


326    }


327    if (secondKey != null) {


328      final length = secondKey.length;


329      final upNum = length > 2 ? pow(10, length).toInt() : 60;


330      final time = duration.inSeconds.remainder(upNum);


331      map[secondKey] = _toDigits(time, length);


332      duration = duration - Duration(seconds: time);


333    }


334    if (millisecondKey != null) {


335      final length = millisecondKey.length;


336      map[millisecondKey] = _toDigits(duration.inMilliseconds, length);


337    }


338    return map;


339  }
