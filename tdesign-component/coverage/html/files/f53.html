152  @override


153  void initState() {


154    super.initState();

161  /// 把group中配置的默认选中id，同步到状态中


162  void _syncCheckState(List<String>? checkIds) {


163    checkBoxStates.clear();


164    checkIds?.forEach((element) {


165      checkBoxStates[element] = true;


166    });


167  }

172    super.didUpdateWidget(oldWidget);


173    final oldCheckIds = oldWidget.value;


174    final newCheckIds = widget.value;


175    if (oldCheckIds != newCheckIds) {


176      _syncCheckState(newCheckIds);


177    }


178  }

180


181  ///


182  /// 根据id获取CheckBox的勾选状态


183  ///


184  ///


185  bool getCheckBoxStateById(String id, bool checked) {


186    if (checkBoxStates[id] == null) {


187      // checkBox本身的状态


188      checkBoxStates[id] = checked;


189    }


190    return checkBoxStates[id]!;


191  }


192


193  /// 勾选单个CheckBox


194  bool toggle(String id, bool check, [bool notify = false]) {


195    // 检查是否超过用户设置的最大可勾选数


196    if (widget.maxChecked != null && check) {


197      if (checkBoxStates.count((k, v) => v) >= widget.maxChecked!) {


198        widget.onOverloadChecked?.call();


199        return false;


200      }


201    }


202    checkBoxStates[id] = check;


203    if (notify) {


204      setState(() {});


205    }


206    _notifyChange();


207    return true;


208  }


209


210  /// 操作所有CheckBox


211  void toggleAll(bool check, [bool notify = true]) {


212    var isChanged = false;


213    checkBoxStates.forEachCanBreak((k, v) {


214      if (check) {


215        if (!toggle(k, check)) {


216          // 勾选失败，退出循环


217          return true;


218        }


219      } else {


220        toggle(k, check);


221      }


222      isChanged = true;


223      return false;


224    });


225


226    if (isChanged && notify) {


227      setState(() {});


228      _notifyChange();


229    }


230  }


231


232  /// 反选


233  void _reverseAll() {


234    final reverseValue =


235    checkBoxStates.map((key, value) => MapEntry(key, !value));


236    checkBoxStates.forEach((key, value) {


237      checkBoxStates[key] = false;


238    });


239    checkBoxStates.forEach((k, v) {


240      var check = reverseValue[k] ?? false;


241      toggle(k, check);


242    });


243    setState(() {});


244    _notifyChange();


245  }


246


247  void _notifyChange() {


248    final change = widget.onChanged;


249    if (change != null) {


250      final value = checkBoxStates.where((k, v) => v).keys.toList();


251      change.call(value);


252    }


253  }

255  @override


256  Widget build(BuildContext context) {


257    return TCheckboxGroupInherited(this, widget.child);


258  }
