436  // _currentTabOffsets[index] is the offset of the start edge of the tab at index, and


437  // _currentTabOffsets[_currentTabOffsets.length] is the end edge of the last tab.


438  int get maxTabIndex => _currentTabOffsets!.length - 2;


439


440  double centerOf(int tabIndex) {


441    assert(_currentTabOffsets != null);


442    assert(_currentTabOffsets!.isNotEmpty);


443    assert(tabIndex >= 0);


444    assert(tabIndex <= maxTabIndex);


445    return (_currentTabOffsets![tabIndex] + _currentTabOffsets![tabIndex + 1]) /


446        2.0;


447  }


448


449  Rect indicatorRect(Size tabBarSize, int tabIndex) {


450    assert(_currentTabOffsets != null);


451    assert(_currentTextDirection != null);


452    assert(_currentTabOffsets!.isNotEmpty);


453    assert(tabIndex >= 0);


454    assert(tabIndex <= maxTabIndex);


455    double tabLeft, tabRight;


456    switch (_currentTextDirection!) {


457      case TextDirection.rtl:


458▶         tabLeft = _currentTabOffsets![tabIndex + 1];


459▶         tabRight = _currentTabOffsets![tabIndex];


460        break;


461      case TextDirection.ltr:


462        tabLeft = _currentTabOffsets![tabIndex];


463        tabRight = _currentTabOffsets![tabIndex + 1];


464        break;


465    }


466


467    if (indicatorSize == TabBarIndicatorSize.label) {


468▶       final tabWidth = tabKeys[tabIndex].currentContext!.size!.width;


469▶       final delta = ((tabRight - tabLeft) - tabWidth) / 2.0;


470▶       tabLeft += delta;


471▶       tabRight -= delta;


472    }


473


474    final insets = indicatorPadding.resolve(_currentTextDirection);


475    final rect =


476        Rect.fromLTWH(tabLeft, 0.0, tabRight - tabLeft, tabBarSize.height);


477


478    if (!(rect.size >= insets.collapsedSize)) {


479▶       throw FlutterError(


480        'indicatorPadding insets should be less than Tab Size\n'


481▶         'Rect Size : ${rect.size}, Insets: ${insets.toString()}',


482      );


483    }


484    return insets.deflateRect(rect);


485  }

487  @override


488  void paint(Canvas canvas, Size size) {


489    _needsPaint = false;


490    _painter ??= indicator.createBoxPainter(markNeedsPaint);


491


492    final index = controller.index.toDouble();


493    final value = controller.animation!.value;


494    final ltr = index > value;


495    final from = (ltr ? value.floor() : value.ceil()).clamp(0, maxTabIndex);


496    final to = (ltr ? from + 1 : from - 1).clamp(0, maxTabIndex);


497    final fromRect = indicatorRect(size, from);


498    final toRect = indicatorRect(size, to);


499    _currentRect = Rect.lerp(fromRect, toRect, (value - from).abs());


500    assert(_currentRect != null);


501


502    final configuration = ImageConfiguration(


503      size: _currentRect!.size,


504      textDirection: _currentTextDirection,


505    );


506    _painter!.paint(canvas, _currentRect!.topLeft, configuration);


507  }

520double _indexChangeProgress(TabController controller) {


521  final controllerValue = controller.animation!.value;


522  final previousIndex = controller.previousIndex.toDouble();


523  final currentIndex = controller.index.toDouble();


524


525  // The controller's offset is changing because the user is dragging the


526  // TabBarView's PageView to the left or right.


527  if (!controller.indexIsChanging) {


528▶     return (currentIndex - controllerValue).abs().clamp(0.0, 1.0);


529  }


530


531  // The TabController animation's value is changing from previousIndex to currentIndex.


532  return (controllerValue - currentIndex).abs() /


533      (currentIndex - previousIndex).abs();

1439▶   bool _handleScrollNotification(ScrollNotification notification) {


1440▶     if (_warpUnderwayCount > 0) {
