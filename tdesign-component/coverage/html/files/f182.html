99class _TStepsState extends State<TSteps> {


100  int _clampActiveIndex(int index, int length) {


101    if (index < 0) {


102      return 0;


103    }


104    if (index >= length) {


105      return length > 0 ? length - 1 : 0;


106    }


107    return index;


108  }


109


110  @override


111  Widget build(BuildContext context) {


112    final effectiveIndex = widget.value ?? widget.activeIndex;


113    final effectiveStatus = widget.status;


114    final effectiveSimple = widget.simple;


115    final effectiveReadOnly = widget.readOnly;


116    final effectiveVerticalSelect = widget.verticalSelect;


117


118    /// 当前激活的step索引


119    final currentActiveIndex = _clampActiveIndex(


120      effectiveIndex,


121      widget.steps.length,


122    );


123


124    return widget.direction == TStepsDirection.horizontal


125        ? TStepsHorizontal(


126            steps: widget.steps,


127            activeIndex: currentActiveIndex,


128            status: effectiveStatus,


129            simple: effectiveSimple,


130            readOnly: effectiveReadOnly)


131        : TStepsVertical(


132            steps: widget.steps,


133            activeIndex: currentActiveIndex,


134            status: effectiveStatus,
