268  /// 创建路由


269  static void _createRoute(


270    BuildContext context, {


271    required TActionSheetTheme theme,


272    required List<TActionSheetItem> items,


273    TActionSheetAlign? align,


274    String? cancelText,


275    bool? showCancel,


276    TActionSheetOnChanged? onChanged,


277    bool? showOverlay,


278    bool? closeOnOverlayClick,


279    int? count,


280    int? rows,


281    double? itemHeight,


282    double? itemMinWidth,


283    bool? scrollable,


284    bool? showPagination,


285    VoidCallback? onCancel,


286    String? subtitle,


287    VoidCallback? onClose,


288    bool? useSafeArea,


289  }) {


290    if (_actionSheetHandle?.isShowing == true) {


291      return;


292    }


293


294    // P1: 组件级 ThemeExtension 回退


295    final tTheme = Theme.of(context).extension<TActionSheetThemeData>();


296    final effectiveAlign = align ?? tTheme?.defaultAlign ?? TActionSheetAlign.center;


297    final effectiveShowCancel = showCancel ?? tTheme?.showCancelButton ?? true;


298    final effectiveShowOverlay = showOverlay ?? true;


299    final effectiveCloseOnOverlayClick = closeOnOverlayClick ?? true;


300    final effectiveCount = count ?? tTheme?.count ?? 8;


301    final effectiveRows = rows ?? tTheme?.rows ?? 2;


302    final effectiveItemHeight = itemHeight ?? tTheme?.itemHeight ?? 96.0;


303    final effectiveItemMinWidth = itemMinWidth ?? tTheme?.itemMinWidth ?? 80.0;


304    final effectiveScrollable = scrollable ?? tTheme?.scrollable ?? false;


305    final effectiveShowPagination = showPagination ?? tTheme?.showPagination ?? false;


306    final effectiveUseSafeArea = useSafeArea ?? tTheme?.useSafeArea ?? true;


307    final effectiveCancelText = cancelText ?? tTheme?.cancelText ?? context.resource.cancel;


308


309    Widget sheetChild;


310    switch (theme) {


311      case TActionSheetTheme.list:


312        sheetChild = TActionSheetList(


313          items: items,


314          align: effectiveAlign,


315          cancelText: effectiveCancelText,


316          subtitle: subtitle,


317          showCancel: effectiveShowCancel,


318          onCancel: onCancel,


319          onChanged: onChanged,


320          useSafeArea: effectiveUseSafeArea,


321        );


322        break;


323      case TActionSheetTheme.grid:


324        sheetChild = TActionSheetGrid(


325          items: items,


326          align: effectiveAlign,


327          onChanged: onChanged,


328          showCancel: effectiveShowCancel,


329          showPagination: effectiveShowPagination,


330          scrollable: effectiveScrollable,


331          cancelText: effectiveCancelText,


332          subtitle: subtitle,


333          count: effectiveCount,


334          rows: effectiveRows,


335          onCancel: onCancel,


336          itemHeight: effectiveItemHeight,


337          itemMinWidth: effectiveItemMinWidth,


338          useSafeArea: effectiveUseSafeArea,


339        );


340        break;


341      case TActionSheetTheme.group:


342        sheetChild = TActionSheetGroup(


343          items: items,


344          align: effectiveAlign,


345          cancelText: effectiveCancelText,


346          showCancel: effectiveShowCancel,


347          onCancel: onCancel,


348          onChanged: onChanged,


349          itemHeight: effectiveItemHeight,


350          itemMinWidth: effectiveItemMinWidth,


351          useSafeArea: effectiveUseSafeArea,


352        );


353        break;


354    }


355


356    _actionSheetHandle = TPopup.show(


357      context,


358      options: TPopupOptions.bottom(


359        cancelBuilder: null,


360        confirmBuilder: null,


361        showOverlay: effectiveShowOverlay,


362        closeOnOverlayClick: effectiveShowOverlay && effectiveCloseOnOverlayClick,


363        overlayColor: effectiveShowOverlay ? null : Colors.transparent,


364        onClosed: onClose,


365        child: sheetChild,


366      ),


367    );


368  }
