282  /// 关闭指定的Toast


283  static void dismissToast(String toastId) {


284    final instance = _toastInstances[toastId];


285    if (instance != null) {


286      instance.cancel();


287      _toastInstances.remove(toastId);


288    }


289  }

291  /// 关闭所有Toast


292  static void dismissAll() {


293    for (final instance in _toastInstances.values) {


294      instance.cancel();


295    }


296    _toastInstances.clear();


297  }

299  /// 关闭加载Toast（向后兼容）


300  static void dismissLoading() {


301    // 关闭所有类型为loading的Toast


302    final loadingIds = _toastInstances.entries


303        .where((entry) => entry.key.startsWith('toast_'))


304        .map((entry) => entry.key)


305        .toList();


306    


307    for (final id in loadingIds) {


308      dismissToast(id);


309    }


310  }

312  static void _showOverlay(


313    Widget? widget, {


314    required BuildContext context,


315    Duration duration = const Duration(milliseconds: 3000),


316    bool? preventTap,


317    required String toastId,


318  }) {


319    // 不自动关闭之前的Toast，支持多个Toast同时显示


320    final overlayState = Overlay.of(context);


321    


322    OverlayEntry overlayEntry;


323    if (preventTap ?? false) {


324      overlayEntry = OverlayEntry(


325        builder: (BuildContext context) => Positioned(


326          top: 0,


327          right: 0,


328          bottom: 0,


329          left: 0,


330          child: Container(


331            color: Colors.transparent,


332            child: Align(


333              alignment: Alignment.center,


334              child: widget,


335            ),


336          ),


337        ),


338      );


339    } else {


340      overlayEntry = OverlayEntry(


341        builder: (BuildContext context) => Center(


342          child: widget,


343        ),


344      );


345    }


346


347    overlayState.insert(overlayEntry);


348


349    Timer? timer;


350    Timer? disposeTimer;


351


352    if (duration != const Duration(seconds: 99999999)) {


353      timer = Timer(duration, () {


354        final instance = _toastInstances[toastId];


355        if (instance != null && instance.showing) {


356          instance.showing = false;


357          overlayEntry.markNeedsBuild();


358          


359          disposeTimer = Timer(const Duration(milliseconds: 200), () {


360            overlayEntry.remove();


361            _toastInstances.remove(toastId);


362          });


363        }


364      });


365    }


366


367    _toastInstances[toastId] = _ToastInstance(


368      overlayEntry: overlayEntry,


369      timer: timer,


370      disposeTimer: disposeTimer,


371    );


372  }


373}
