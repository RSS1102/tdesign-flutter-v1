37  List<String> _getWeeks() {


38    final ans = <String>[];


39    var i = firstDayOfWeek % 7;


40    while (ans.length < 7) {


41      ans.add(weekdayNames[i]);


42      i = (i + 1) % 7;


43    }


44    return ans;


45  }

55            if (index != 0) SizedBox(width: weekdayGap),
