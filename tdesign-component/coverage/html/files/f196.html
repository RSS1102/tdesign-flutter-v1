18/// An iOS-style switch.


19///


20/// Used to toggle the on/off state of a single setting.


21///


22/// The switch itself does not maintain any state. Instead, when the state of


23/// the switch changes, the widget calls the [onChanged] callback. Most widgets


24/// that use a switch will listen for the [onChanged] callback and rebuild the


25/// switch with a new [value] to update the visual appearance of the switch.


26///


27/// {@tool dartpad}


28/// This example shows a toggleable [TCupertinoSwitch]. When the thumb slides to


29/// the other side of the track, the switch is toggled between on/off.


30///


31/// ** See code in examples/api/lib/cupertino/switch/cupertino_switch.0.dart **


32/// {@end-tool}


33///


34/// {@tool snippet}


35///


36/// This sample shows how to use a [TCupertinoSwitch] in a [ListTile]. The


37/// [MergeSemantics] is used to turn the entire [ListTile] into a single item


38/// for accessibility tools.


39///


40/// ```dart


41/// MergeSemantics(


42///   child: ListTile(


43///     title: const Text('Lights'),


44///     trailing: TCupertinoSwitch(


45///       value: _lights,


46///       onChanged: (bool value) { setState(() { _lights = value; }); },


47///     ),


48///     onTap: () { setState(() { _lights = !_lights; }); },


49///   ),


50/// )


51/// ```


52/// {@end-tool}


53///


54/// See also:


55///


56///  * [Switch], the Material Design equivalent.


57///  * <https://developer.apple.com/ios/human-interface-guidelines/controls/switches/>

122  /// {@template flutter.cupertino.TCupertinoSwitch.dragStartBehavior}


123  /// Determines the way that drag start behavior is handled.


124  ///


125  /// If set to [DragStartBehavior.start], the drag behavior used to move the

209▶   @override

211▶     super.didUpdateWidget(oldWidget);


212▶     _drag.dragStartBehavior = widget.dragStartBehavior;

214▶     if (needsPositionAnimation || oldWidget.value != widget.value) {

219  // `isLinear` must be true if the position animation is trying to move the


220  // thumb to the closest end after the most recent drag animation, so the curve


221  // does not change when the controller's value is not 0 or 1.


222  //


223  // It can be set to false when it's an implicit animation triggered by


224  // widget.value changes.


225▶   void _resumePositionAnimation({bool isLinear = true}) {


226▶     needsPositionAnimation = false;


227▶     position


228▶       ..curve = isLinear ? Curves.linear : Curves.ease


229▶       ..reverseCurve = isLinear ? Curves.linear : Curves.ease.flipped;


230▶     if (widget.value) {


231▶       _positionController.forward();


232    } else {


233▶       _positionController.reverse();


234    }


235  }

237  void _handleTapDown(TapDownDetails details) {


238    if (isInteractive) {


239      needsPositionAnimation = false;


240    }


241    _reactionController.forward();


242  }

244  void _handleTap() {


245    if (isInteractive) {


246      widget.onChanged!(!widget.value);


247      _emitVibration();


248    }


249  }

251  void _handleTapUp(TapUpDetails details) {


252    if (isInteractive) {


253      needsPositionAnimation = false;


254      _reactionController.reverse();


255    }


256  }


257


258▶   void _handleTapCancel() {


259▶     if (isInteractive) {


260▶       _reactionController.reverse();


261    }


262  }


263


264▶   void _handleDragStart(DragStartDetails details) {


265▶     if (isInteractive) {


266▶       needsPositionAnimation = false;


267▶       _reactionController.forward();


268▶       _emitVibration();


269    }


270  }


271


272▶   void _handleDragUpdate(DragUpdateDetails details) {


273▶     if (isInteractive) {


274▶       position


275▶         ..curve = Curves.linear


276▶         ..reverseCurve = Curves.linear;


277▶       final delta = details.primaryDelta! / _kTrackInnerLength;


278▶       switch (Directionality.of(context)) {


279▶         case TextDirection.rtl:


280▶           _positionController.value -= delta;


281          break;


282▶         case TextDirection.ltr:


283▶           _positionController.value += delta;


284          break;


285      }


286    }


287  }


288


289▶   void _handleDragEnd(DragEndDetails details) {


290    // Deferring the animation to the next build phase.


291▶     setState(() {


292▶       needsPositionAnimation = true;


293    });


294    // Call onChanged when the user's intent to change value is clear.


295▶     if (position.value >= 0.5 != widget.value) {


296▶       widget.onChanged!(!widget.value);


297    }


298▶     _reactionController.reverse();


299  }

301  void _emitVibration() {


302    switch (defaultTargetPlatform) {


303      case TargetPlatform.iOS:


304▶         HapticFeedback.lightImpact();


305        break;


306      default:


307        break;


308    }


309  }

407const double _kTrackWidth = 45.0;


408const double _kTrackHeight = 28.0;


409const double _kTrackRadius = _kTrackHeight / 2.0;


410const double _kTrackInnerStart = _kTrackHeight / 2.0;


411const double _kTrackInnerEnd = _kTrackWidth - _kTrackInnerStart;


412const double _kTrackInnerLength = _kTrackInnerEnd - _kTrackInnerStart;


413const double _kSwitchWidth = 45.0;


414const double _kSwitchHeight = 28.0;


415// Opacity of a disabled switch, as eye-balled from iOS Simulator on Mac.


416const double _kTDCupertinoSwitchDisabledOpacity = 0.5;


417// In the normal switch case, the closed thumb is smaller than the open thumb.

419const double _kSwitchOnThumbRadius = 11;


420const double _kSwitchOffThumbRadius = 8;


421const double _kSwitchOnThumbMargin = (_kTrackHeight - _kSwitchOnThumbRadius * 2) / 2;


422const double _kSwitchOffThumbMargin = (_kTrackHeight - _kSwitchOffThumbRadius * 2) / 2;

563  @override


564  void paint(PaintingContext context, Offset offset) {


565    final canvas = context.canvas;


566


567    final currentValue = _state.position.value;


568    final currentReactionValue = _state._reaction.value;


569


570    final double visualPosition;


571    switch (textDirection) {


572      case TextDirection.rtl:


573▶         visualPosition = 1.0 - currentValue;


574        break;


575      case TextDirection.ltr:


576        visualPosition = currentValue;


577        break;


578    }


579


580    final paint = Paint()


581      ..color = Color.lerp(trackColor, activeColor, currentValue)!;


582


583    final trackRect = Rect.fromLTWH(


584      offset.dx + (size.width - _kTrackWidth) / 2.0,


585      offset.dy + (size.height - _kTrackHeight) / 2.0,


586      _kTrackWidth,


587      _kTrackHeight,


588    );


589    final trackRRect = RRect.fromRectAndRadius(


590        trackRect, const Radius.circular(_kTrackRadius));


591    canvas.drawRRect(trackRRect, paint);


592


593    final currentThumbExtension =


594        CupertinoThumbPainter.extension * currentReactionValue;


595    final thumbLeft = lerpDouble(


596      trackRect.left + _kTrackInnerStart - thumbRadius,


597      trackRect.left + _kTrackInnerEnd - thumbRadius - currentThumbExtension,


598      visualPosition,


599    )!;


600    final thumbRight = lerpDouble(


601      trackRect.left + _kTrackInnerStart + thumbRadius + currentThumbExtension,


602      trackRect.left + _kTrackInnerEnd + thumbRadius,


603      visualPosition,


604    )!;


605    final thumbCenterY = offset.dy + size.height / 2.0;


606    final thumbBounds = Rect.fromLTRB(


607      thumbLeft,


608      thumbCenterY - thumbRadius,


609      thumbRight,


610      thumbCenterY + thumbRadius,


611    );


612


613    _clipRRectLayer.layer = context


614        .pushClipRRect(needsCompositing, Offset.zero, thumbBounds, trackRRect,


615            (PaintingContext innerContext, Offset offset) {


616      _thumbPainter.paint(innerContext.canvas, thumbBounds);


617    }, oldLayer: _clipRRectLayer.layer);


618    if (child != null) {


619      context.paintChild(child!, Offset(thumbBounds.left + thumbMargin, 0));


620    }


621  }
