6/// TIcon — v1.0 图标组件


7///


8/// Material [Icon] 的薄包装（T2 纯展示），提供组件级 Theme 注入能力。


9/// 图标数据由 `tdesign_icons` 资源包提供，通过 `TIcons.xxx` 常量引用。


10///


11/// 优先级链：构造器参数 > [TIconThemeData] > [IconTheme]


12///


13/// 示例：


14/// ```dart


15/// // 基础使用


16/// TIcon(TIcons.home_filled)


17///


18/// // 指定尺寸和颜色


19/// TIcon(TIcons.setting, size: 24, color: Colors.blue)


20///


21/// // 通过名称引用


22/// TIcon.fromName('home_filled')


23///


24/// // 子树 Theme 注入


25/// Theme(


26///   data: Theme.of(context).mergeExtension(


27///     const TIconThemeData(size: 20, color: Colors.grey),


28///   ),


29///   child: TIcon(TIcons.home_filled),


30/// )


31/// ```


32class TIcon extends StatelessWidget {

38

40  final Color? color;


41

44

50    this.semanticLabel,

53  /// 通过图标名称构造（查找 [TIcons.allIconsMap]）


54  ///


55  /// 如果名称不存在，抛出 [ArgumentError]。


56  factory TIcon.fromName(


57    String name, {


58    Key? key,


59    double? size,


60    Color? color,


61    String? semanticLabel,


62  }) {


63    final iconData = TIcons.allIconsMap[name];


64    if (iconData == null) {


65      throw ArgumentError('Unknown icon name: $name');


66    }


67    return TIcon(


68      iconData,


69      key: key,


70      size: size,


71      color: color,


72      semanticLabel: semanticLabel,


73    );


74  }

76  @override

83    final effectiveColor =
