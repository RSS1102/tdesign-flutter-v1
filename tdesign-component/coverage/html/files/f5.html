34/// 弹出层入口：五向滑入 / 居中弹出，支持蒙层、bottom 操作区、center 面板外下方关闭区。


35///


36/// 通过 [show] 命令式打开；返回 [TPopupHandle] 用于关闭与再次打开。


37/// 多次调用 [show] 会继续压入新的浮层路由，可用于叠加展示。


38///


39/// **示例**


40///


41/// ```dart


42/// final handle = TPopup.show(


43///   context,


44///   options: TPopupOptions.bottom(


45///     titleWidget: const Text('标题'),


46///     child: MyPanel(),


47///   ),


48/// );


49/// handle.close();


50/// handle.open();


51/// ```


52///


53/// 配置项见 [TPopupOptions]；方向见 [TPopupPlacement]。


54final class TPopup {


55▶   const TPopup._();


56


57  /// 打开浮层并压入独立 [PopupRoute]。


58  ///


59  /// [context] 用于查找 [Navigator] 并展示浮层。


60  ///


61  /// [options] 浮层配置；方向固定时推荐 [TPopupOptions.bottom] 等命名工厂。


62  ///


63  /// 返回 [TPopupHandle]，可用 [TPopupHandle.close]、[TPopupHandle.open]、


64  /// [TPopupHandle.isShowing] 控制与查询。


65  /// 重复调用会继续 push 新的浮层；若需互斥请在业务层管理。


66  ///


67  /// [navigatorContext] 可选，指定承载浮层的 [Navigator] 的 context，默认 [context]。


68  ///


69  /// [useRootNavigator] 为 true 时使用根 [Navigator]（嵌套导航场景）。


70  static TPopupHandle show(


71    BuildContext context, {


72    required TPopupOptions options,


73    BuildContext? navigatorContext,


74    bool useRootNavigator = false,


75  }) {


76    final navContext = navigatorContext ?? context;


77    final handle = TPopupHandle._(


78      options: options,


79      navigatorContext: navigatorContext,


80      useRootNavigator: useRootNavigator,


81    );


82    handle.open(navContext);


83    return handle;


84  }


85}
