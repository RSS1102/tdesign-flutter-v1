23  @override

55  @override

64      children: [


65        if (showCancel)


66          Padding(

68            child: _wrapDefaultCancelSemantics(


69              context: context,


70              child: _buildCancel(context, theme),


71            ),


72          )


73        else


74          SizedBox(width: theme.spacer16),


75        Expanded(


76          child: title == null

80        if (showConfirm)


81          Padding(


82            padding: EdgeInsets.only(right: theme.spacer8),


83            child: _wrapDefaultConfirmSemantics(


84              context: context,

94  Widget _wrapDefaultCancelSemantics({

98    if (!_isPopupDefaultCancel(options.cancelBuilder)) {

139    if (_isPopupDefaultCancel(options.cancelBuilder)) {

154


155  Widget _buildConfirm(BuildContext context, TThemeData theme) {
