92  Widget _buildButton(BuildContext context) {


93    if (buttonWidget != null) {


94      return buttonWidget!;


95    }


96    if (buttonStyle == TDialogButtonStyle.text) {


97      return Column(


98        mainAxisSize: MainAxisSize.min,


99        children: [


100          const SizedBox(height: 23),


101          const TDivider(),


102          TDialogButton(


103            buttonText: buttonText ?? context.resource.knew,


104            buttonTextColor: buttonTextColor,


105            buttonVariant: TButtonVariant.text,


106            buttonColorScheme: TButtonColorScheme.primary,


107            height: 56,


108            buttonStyle: buttonStyleCustom,


109▶             onPressed: () {


110▶               if (onPressed != null) {


111▶                 onPressed!();


112              } else {


113▶                 Navigator.pop(context);


114              }


115            },


116          )


117        ],


118      );


119    } else {


120      return Container(


121        padding: const EdgeInsets.fromLTRB(24, 24, 24, 24),


122        child: TDialogButton(


123          buttonText: buttonText ?? context.resource.knew,


124          buttonTextColor: buttonTextColor,


125          buttonColorScheme: TButtonColorScheme.primary,


126          buttonStyle: buttonStyleCustom,


127          onPressed: () {


128            if (onPressed != null) {


129              onPressed!();


130            } else {


131              Navigator.pop(context);


132            }


133          },


134        ),


135      );


136    }


137  }
