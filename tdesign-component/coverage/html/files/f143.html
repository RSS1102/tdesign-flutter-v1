6/// 自由结构数据 → 规范化 `TPickerOption` 树的归一化工具


7///


8/// `TPickerColumns.fromRaw` / `TPickerLinked.fromRaw` 在构造时调用本类的


9/// 静态方法把 `List` / `Map` 转成 `List<List<TPickerOption>>` 或


10/// `Map<TPickerOption, dynamic>`，让 `TPicker` 内部只需面对一种数据形态。


11///


12/// 类型短路：若传入的 `raw` 已经是目标类型，**直接返回**而不深拷贝，


13/// 避免在父级每帧用 `.fromRaw(...)` 构造 items 时产生大量临时对象。


14@internal


15class TPickerNormalize {


16▶   TPickerNormalize._();


17


18  /// 归一化多列独立数据


19  ///


20  /// - 已是 `List<List<TPickerOption>>` 时直接返回（无拷贝）。


21  /// - 否则按 `keys` 字段映射每个元素为 [TPickerOption]；非 `List` 列


22  ///   退化为空列（保留列数，避免下游索引越界）。


23  static List<List<TPickerOption>> normalizeColumns(


24      List rawColumns, TPickerKeys keys) {


25    if (rawColumns is List<List<TPickerOption>>) {


26      return rawColumns;


27    }


28    return [


29      for (final col in rawColumns)


30        if (col is List)


31          [for (final raw in col) _toOption(raw, keys)]


32        else


33          <TPickerOption>[],


34    ];


35  }


36


37  /// 归一化联动树数据


38  ///


39  /// - 已是 `Map<TPickerOption, dynamic>` 时直接返回（无拷贝）。


40  /// - 否则按 `keys` 逐层下钻：value 为 `Map` 视为下一级联动，为


41  ///   `List` 视为叶子级选项；非 `Map`/`List` 退化为空叶子。


42  static Map<TPickerOption, dynamic> normalizeLinked(


43      Map rawTree, TPickerKeys keys) {


44    if (rawTree is Map<TPickerOption, dynamic>) {


45      return rawTree;


46    }


47    return _convertLinkedMap(rawTree, keys);


48  }


49


50  /// 单条 `raw` → [TPickerOption] 的转换


51  ///


52  /// 优先级：已是 [TPickerOption] → 透传；是 `Map` → 按 `keys` 读字段；


53  /// 其他 → `toString()` 作为 label，原值作为 value。


54  static TPickerOption _toOption(dynamic raw, TPickerKeys keys) {


55    if (raw is TPickerOption) {


56      return raw;


57    }


58    if (raw is Map) {


59      final label = raw[keys.label]?.toString() ?? '';


60      final value = raw[keys.value];


61      final disabledRaw = raw[keys.disabled];


62      final disabled = disabledRaw is bool ? disabledRaw : false;


63      return TPickerOption(label: label, value: value, disabled: disabled);


64    }


65    return TPickerOption(label: raw?.toString() ?? '', value: raw);


66  }


67


68  static Map<TPickerOption, dynamic> _convertLinkedMap(


69      Map src, TPickerKeys keys) {


70    final result = <TPickerOption, dynamic>{};


71    src.forEach((rawKey, rawVal) {


72      final keyOpt = _toOption(rawKey, keys);


73      result[keyOpt] = _convertChild(rawVal, keys);


74    });


75    return result;


76  }


77


78  static dynamic _convertChild(dynamic child, TPickerKeys keys) {


79    if (child is Map) {


80      return _convertLinkedMap(child, keys);


81    }


82    if (child is List) {


83      return [for (final r in child) _toOption(r, keys)];


84    }


85    return const <TPickerOption>[];


86  }


87}
