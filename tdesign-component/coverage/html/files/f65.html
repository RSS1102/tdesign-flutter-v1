37/// 在 `[start, end]` 与当前选中上下文下，计算单列的 min/max（闭区间）。


38_IntRange columnBounds(


39  DateTimeColumn col, {


40  required DateTime current,


41  required DateTime? start,


42  required DateTime? end,


43  required int yearAnchor,


44  required List<DateTimeColumn> columns,


45}) {


46  final safeEnd = _safeEnd(start, end);


47


48  if (_isTimeOnlyColumns(columns)) {


49    return _timeOnlyColumnBounds(col,


50        current: current, start: start, end: safeEnd);


51  }


52


53  return switch (col) {


54    DateTimeColumn.year => _IntRange(


55        start?.year ?? (yearAnchor - DateTimePickerSnapshot.defaultYearOffset),


56        safeEnd?.year ??


57            (yearAnchor + DateTimePickerSnapshot.defaultYearOffset),


58      ),


59    DateTimeColumn.month => _IntRange(


60        (start != null && current.year == start.year) ? start.month : 1,


61        (safeEnd != null && current.year == safeEnd.year) ? safeEnd.month : 12,


62      ),


63    DateTimeColumn.day => () {


64        final maxDay = DateTimePickerSnapshot.daysInMonth(


65          current.year,


66          current.month,


67        );


68        final startDay =


69            (start != null && _isSameMonth(current, start)) ? start.day : 1;


70        final endDay = (safeEnd != null && _isSameMonth(current, safeEnd))


71            ? safeEnd.day.clamp(1, maxDay)


72            : maxDay;


73        return _IntRange(startDay, endDay);


74      }(),


75    DateTimeColumn.hour => _IntRange(


76        (start != null && _isSameDate(current, start)) ? start.hour : 0,


77        (safeEnd != null && _isSameDate(current, safeEnd)) ? safeEnd.hour : 23,


78      ),


79    DateTimeColumn.minute => _IntRange(


80        (start != null && _isSameHour(current, start)) ? start.minute : 0,


81        (safeEnd != null && _isSameHour(current, safeEnd))


82            ? safeEnd.minute


83            : 59,


84      ),


85    DateTimeColumn.second => _IntRange(


86        (start != null && _isSameMinute(current, start)) ? start.second : 0,


87        (safeEnd != null && _isSameMinute(current, safeEnd))


88            ? safeEnd.second


89            : 59,


90      ),


91  };

114int _firstStepValue(int min, int step) {


115  if (step <= 1) {


116    return min;


117  }


118  final rem = min % step;


119  return rem == 0 ? min : min + (step - rem);


120}


121


122▶ int _lastStepValue(int min, int max, int step) {


123▶   if (step <= 1) {


124    return max;


125  }


126▶   final first = _firstStepValue(min, step);


127▶   if (first > max) {


128    return min;


129  }


130▶   final count = (max - first) ~/ step;


131▶   return first + count * step;


132}

134int snapToStep(int value, int min, int max, int step) {


135  if (max < min) {


136    return min;


137  }


138  if (step <= 1) {


139    return value.clamp(min, max);


140  }


141  if (value < min) {


142▶     return _firstStepValue(min, step);


143  }


144  if (value > max) {


145▶     return _lastStepValue(min, max, step);


146  }


147  final first = _firstStepValue(min, step);


148  final snapped = first + (((value - first) / step).round()) * step;


149  if (snapped > max) {


150▶     return _lastStepValue(min, max, step);


151  }


152  if (snapped < min) {


153    return first;


154  }


155  return snapped;


156}

158DateTime normalizePickerDateTime(


159  DateTime dt, {


160  required List<DateTimeColumn> columns,


161  DateTime? start,


162  DateTime? end,


163  DateTimePickerSteps? steps,


164  required int yearAnchor,


165}) {


166  final safeEnd = _safeEnd(start, end);


167  var result =


168      DateTimePickerSnapshot.clampDateTime(dt, start: start, end: safeEnd);


169


170  var y = result.year;


171  var m = result.month;


172  var d = result.day;


173  var h = result.hour;


174  var mi = result.minute;


175  var s = result.second;


176


177  for (final col in columns) {


178    final bounds = columnBounds(


179      col,


180      current: result,


181      start: start,


182      end: safeEnd,


183      yearAnchor: yearAnchor,


184      columns: columns,


185    );


186    if (!bounds.isValid) {


187      continue;


188    }


189    final step = steps?.forColumn(col) ?? 1;


190    switch (col) {


191      case DateTimeColumn.year:


192        y = snapToStep(y, bounds.min, bounds.max, step);


193      case DateTimeColumn.month:


194        m = snapToStep(m, bounds.min, bounds.max, step);


195      case DateTimeColumn.day:


196        d = snapToStep(d, bounds.min, bounds.max, step);


197      case DateTimeColumn.hour:


198        h = snapToStep(h, bounds.min, bounds.max, step);


199      case DateTimeColumn.minute:


200        mi = snapToStep(mi, bounds.min, bounds.max, step);


201      case DateTimeColumn.second:


202        s = snapToStep(s, bounds.min, bounds.max, step);


203    }


204    result = DateTime(y, m, d, h, mi, s);


205  }


206


207  final maxDay = DateTimePickerSnapshot.daysInMonth(y, m);


208  if (d > maxDay) {


209    d = maxDay;


210  }


211  result = DateTime(y, m, d, h, mi, s);


212  return DateTimePickerSnapshot.clampDateTime(result,


213      start: start, end: safeEnd);


214}
