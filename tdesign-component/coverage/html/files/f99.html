65    required Widget? icon,


66    required TButtonProps? buttonProps,


67    required VoidCallback? onPressed,


68    required BuildContext context,


69  }) {


70    final hasText = text != null && text.isNotEmpty;


71    final effectiveIcon = icon ?? const Icon(TFabDefaults.defaultIconData);


72


73    // shape 推导：纯图标=圆形，有文字=胶囊形；buttonProps.shape 覆盖


74    final effectiveShape =


75        buttonProps?.shape ?? (hasText ? TButtonShape.round : TButtonShape.circle);


76


77    final tButton = TButton(


78      child: hasText ? Text(text) : null,


79      icon: effectiveIcon,


80      onPressed: onPressed,


81      size: buttonProps?.size ?? TButtonSize.large,


82      variant: buttonProps?.variant ?? TButtonVariant.fill,


83      colorScheme: buttonProps?.colorScheme ?? TButtonColorScheme.primary,


84      style: buttonProps?.style,


85    );


86


87    // TButton 的 shape 由 TButtonThemeData.shape 控制


88    // 合并父级 TButtonThemeData（保留其他字段）+ 推导出的 shape


89    final parentBtnTheme = Theme.of(context).extension<TButtonThemeData>();


90    final fabBtnTheme =


91        (parentBtnTheme ?? const TButtonThemeData()).copyWith(shape: effectiveShape);


92


93    return Theme(


94      data: Theme.of(context).copyWith(


95        extensions: [


96          ...Theme.of(context).extensions.values


97              .where((e) => e is! TButtonThemeData),


98          fabBtnTheme,


99        ],


100      ),


101      child: tButton,


102    );


103  }
