206  void _scrollToTarget(String newIndex, String oldIndex) {


207    _isAnimating = true;


208


209    /// isUp: 是否（手指）向上滑动


210    final isUp = _indexList.indexOf(newIndex) > _indexList.indexOf(oldIndex);


211    if (isUp) {


212      var index = oldIndex;


213      final contentRenderBox =


214          _contentKeys[index]?.findRenderObject() as RenderBox?;


215      if (contentRenderBox != null) {


216        final contentHeight = contentRenderBox.size.height;


217        final maxScrollExtent = _scrollController.position.maxScrollExtent;


218        final targetOffset = contentRenderBox.localToGlobal(


219            Offset(0, contentHeight),


220            ancestor: context.findRenderObject());


221        final scrollOffset = targetOffset.dy + _scrollController.offset;


222        _scrollController.jumpTo(min(maxScrollExtent, scrollOffset));


223      }


224      index = _indexList[_indexList.indexOf(index) + 1];
