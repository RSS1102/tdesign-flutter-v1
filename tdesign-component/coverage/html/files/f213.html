214    required FontWeight? resolvedFontWeight,

218    // globalFontFamily 注入（替代 v0.2.x 的 kTextNeedGlobalFontFamily 全局变量）


219    final globalFontFamily = configuration?.globalFontFamily;


220    styleFontFamily ??= globalFontFamily?.fontFamily;


221


222    // iOS FontWeight≤w500 且无 fontFamily → 回退 PingFang SC


223    if (PlatformUtil.isIOS &&


224▶         (styleFontFamily == null || styleFontFamily.isEmpty) &&


225        resolvedFontWeight != null &&


226▶         resolvedFontWeight.value <= FontWeight.w500.value) {


227      return 'PingFang SC';


228    }

231  }
