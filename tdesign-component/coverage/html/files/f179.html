308  @override


309  void paint(


310    PaintingContext context,


311    Offset offset, {


312    required RenderBox parentBox,


313    required SliderThemeData sliderTheme,


314    required Animation<double> enableAnimation,


315    required TextDirection textDirection,


316    required Offset thumbCenter,


317    Offset? secondaryOffset,


318    bool isDiscrete = false,


319    bool isEnabled = false,


320    double additionalActiveTrackHeight = 2,

322    assert(sliderTheme.disabledActiveTrackColor != null);


323    assert(sliderTheme.disabledInactiveTrackColor != null);


324    assert(sliderTheme.activeTrackColor != null);


325    assert(sliderTheme.inactiveTrackColor != null);


326    assert(sliderTheme.thumbShape != null);


327    // If the slider [SliderThemeData.trackHeight] is less than or equal to 0,

343      ..color = activeTrackColorTween.evaluate(enableAnimation)!;


344    final inactivePaint = Paint()


345      ..color = inactiveTrackColorTween.evaluate(enableAnimation)!;


346    final Paint leftTrackPaint;


347    final Paint rightTrackPaint;


348    switch (textDirection) {


349      case TextDirection.ltr:


350        leftTrackPaint = activePaint;


351        rightTrackPaint = inactivePaint;


352        break;


353▶       case TextDirection.rtl:


354        leftTrackPaint = inactivePaint;


355        rightTrackPaint = activePaint;


356        break;


357    }


358


359    final trackRect = getPreferredRect(


360      parentBox: parentBox,


361      offset: offset,


362      sliderTheme: sliderTheme,


363      isEnabled: isEnabled,


364      isDiscrete: isDiscrete,


365    );


366    // record size ,Use it when calculating thumb text position


367    themeData.sliderMeasureData.trackerRect = trackRect;


368    final trackRadius = Radius.circular(trackRect.height / 2);


369    final activeTrackRadius =


370        Radius.circular((trackRect.height + additionalActiveTrackHeight) / 2);


371


372    context.canvas.drawRRect(


373      RRect.fromLTRBAndCorners(


374        trackRect.left,


375        (textDirection == TextDirection.rtl)


376▶             ? trackRect.top - (additionalActiveTrackHeight / 2)


377            : trackRect.top,


378        thumbCenter.dx,


379        (textDirection == TextDirection.rtl)


380▶             ? trackRect.bottom + (additionalActiveTrackHeight / 2)


381            : trackRect.bottom,


382        topLeft: (textDirection == TextDirection.ltr)


383            ? activeTrackRadius


384            : trackRadius,


385        bottomLeft: (textDirection == TextDirection.ltr)


386            ? activeTrackRadius

388      ),


389      leftTrackPaint,

391    context.canvas.drawRRect(

394        (textDirection == TextDirection.rtl)


395▶             ? trackRect.top - (additionalActiveTrackHeight / 2)

438  /// The resting elevation adds shadow to the unpressed thumb.

449  /// The default is 6.

456

458  final Color strokeColor;

460  @override


461  Size getPreferredSize(bool isEnabled, bool isDiscrete) {


462    return Size.fromRadius(


463        isEnabled == true ? enabledThumbRadius : _disabledThumbRadius);


464  }


465


466  @override


467  void paint(


468    PaintingContext context,


469    Offset center, {


470    required Animation<double> activationAnimation,


471    required Animation<double> enableAnimation,


472    required bool isDiscrete,


473    required TextPainter labelPainter,


474    required RenderBox parentBox,


475    required SliderThemeData sliderTheme,


476    required TextDirection textDirection,


477    required double value,


478    required double textScaleFactor,


479    required Size sizeWithOverflow,


480  }) {


481    assert(sliderTheme.disabledThumbColor != null);


482    assert(sliderTheme.thumbColor != null);


483


484    final canvas = context.canvas;


485    final radiusTween = Tween<double>(


486      begin: _disabledThumbRadius,


487      end: enabledThumbRadius,


488    );


489    final colorTween = ColorTween(


490      begin: sliderTheme.disabledThumbColor,


491      end: sliderTheme.thumbColor,


492    );


493


494    final color = colorTween.evaluate(enableAnimation)!;


495    final radius = radiusTween.evaluate(enableAnimation);


496


497    final elevationTween = Tween<double>(


498      begin: elevation,


499      end: pressedElevation,


500    );


501


502    final evaluatedElevation = elevationTween.evaluate(activationAnimation);


503    final path = Path()


504      ..addArc(


505          Rect.fromCenter(


506              center: center, width: 2 * radius, height: 2 * radius),


507          0,


508          math.pi * 2);


509


510    var paintShadows = true;


511


512    if (paintShadows) {


513      canvas.drawShadow(


514          path, const Color.fromRGBO(0, 0, 0, 0.5), evaluatedElevation, true);


515    }


516    // draw thumb text


517    if (themeData.showThumbValue &&


518        themeData.sliderMeasureData.trackerRect != null) {


519      var trackerRect = themeData.sliderMeasureData.trackerRect!;


520      var ratio = (center.dx - trackerRect.left) /


521          (trackerRect.right - trackerRect.left);


522      //计算滑块的值


523      var value = (themeData.max - themeData.min) * ratio + themeData.min;


524      //格式化显示


525      var formatterValue = themeData.scaleFormatter == null


526          ? value.toStringAsFixed(2)


527▶           : themeData.scaleFormatter!(value);


528      //绘制数值


529      var painter = TextPainter(


530          text: TextSpan(


531              text: '$formatterValue',


532              style: enableAnimation.value > 0


533                  ? themeData.effectiveThumbTextStyle


534▶                   : themeData.effectiveDisabledThumbTextStyle),


535          textDirection: TextDirection.ltr,


536          textAlign: TextAlign.center)


537        ..layout(maxWidth: 100);


538


539      var textPosition = Offset(


540          center.dx - painter.width / 2, center.dy - painter.height - 14);


541      painter.paint(context.canvas, textPosition);


542      themeData.sliderMeasureData.thumbTextRect = Rect.fromLTWH(


543          center.dx - painter.width / 2,


544          center.dy - painter.height - 14,


545          painter.width,


546          painter.height);


547    }


548    var paint = Paint();


549    paint.color = color;


550    canvas.drawCircle(


551      center,


552      radius,


553      paint,


554    );


555    canvas.drawArc(


556        Rect.fromCircle(center: center, radius: radius),


557        0,


558        2 * math.pi,


559        false,


560        Paint()


561          ..style = PaintingStyle.stroke


562          ..strokeWidth = 1


563          ..color = strokeColor);


564  }
