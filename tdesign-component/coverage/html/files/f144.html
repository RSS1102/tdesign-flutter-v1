121/// 距离到样式的映射器（内部默认渲染与 [ItemBuilderType] 回调共用）


122///


123/// 默认采用 4 档离散赋值（0=选中, 1=紧邻, 2=近边, 3+=远边）。


124/// 在 [ItemBuilderType] 中可继承此类自定义颜色/字号/透明度过渡曲线。


125class ItemDistanceCalculator {


126  const ItemDistanceCalculator();


127


128  /// 4 档粗细：选中 → 最远


129  static const List<FontWeight> _fontWeightLevels = <FontWeight>[


130    FontWeight.w700,


131    FontWeight.w500,


132    FontWeight.w400,


133    FontWeight.w300,


134  ];


135


136  /// 4 档字号缩放因子：选中 → 最远


137  static const List<double> _fontSizeScales = <double>[1.00, 0.94, 0.88, 0.82];


138


139  /// 4 档颜色混合比例（主色 → 占位色）：选中 → 最远


140  static const List<double> _colorMixLevels = <double>[0.00, 0.55, 0.78, 1.00];


141


142  /// 把连续距离量化为 0~3 的档位


143  static int _level(double distance) => distance.round().clamp(0, 3);


144


145  /// 计算指定距离处的文字颜色


146  Color calculateColor(BuildContext context, double distance) {


147    final theme = context.tTheme;


148    final primary = theme.textColorPrimary;


149    final placeholder = theme.textColorPlaceholder;


150    final mix = _colorMixLevels[_level(distance)];


151    if (mix == 0) {


152      return primary;


153    }


154    if (mix >= 1) {


155      return placeholder;


156    }


157    return Color.lerp(primary, placeholder, mix) ?? placeholder;


158  }


159


160  /// 计算指定距离处的文字粗细


161  FontWeight calculateFontWeight(BuildContext context, double distance) =>


162      _fontWeightLevels[_level(distance)];


163


164  /// 计算指定距离处的字体大小


165  double calculateFont(BuildContext context, double distance) {


166    final baseSize = context.tTheme.fontBodyLarge?.size ?? _kBaseFontSize;


167    return baseSize * _fontSizeScales[_level(distance)];


168  }


169


170  /// 计算指定距离处的不透明度（非选中项统一半透明）


171  double calculateOpacity(double distance) =>


172      _level(distance) == 0 ? 1.00 : 0.75;


173}
