76  void updateBadgeNum(String? newCount) {


77    if (newCount == null) {


78      return;


79    }


80    setState(() {


81      // 如果 newCount 超过了 maxCount，则显示 `${maxCount}+`


82      final countValue = int.tryParse(newCount) ?? 0;


83      final maxCountValue = int.tryParse(widget.maxCount ?? '') ?? 0;


84      if (maxCountValue > 0 && countValue > maxCountValue) {


85        badgeNum = '${maxCountValue}+';


86      } else {


87        badgeNum = newCount;


88      }


89    });


90  }


91


92  double getBadgeSize() {


93    switch (widget.size) {


94      case TBadgeSize.large:


95        return 20;


96      case TBadgeSize.small:


97        return 16;


98    }


99  }


100


101  Font? getBadgeFont(BuildContext context) {


102    switch (widget.size) {


103      case TBadgeSize.large:


104        return context.tTheme.fontMarkSmall;


105      case TBadgeSize.small:


106        return context.tTheme.fontMarkExtraSmall;


107    }


108  }


109


110  bool get visible {


111    final theme = _theme(context);


112    final showZero = theme?.showZero ?? true;


113    final parsedValue = double.tryParse(value);


114    return showZero ||


115        (parsedValue != null && parsedValue != 0) ||


116        parsedValue == null;

296class TrapezoidPath extends CustomClipper<Path> {


297  final double widthLarge;


298  final double widthSmall;


299


300  TrapezoidPath(this.widthLarge, this.widthSmall);


301


302  @override


303  Path getClip(Size size) {


304    var path = Path();


305    path.moveTo(0, 0);


306    path.lineTo(widthLarge - widthSmall, 0);


307    path.lineTo(widthLarge, widthSmall);


308    path.lineTo(widthLarge, widthLarge);


309    path.lineTo(0, 0);


310    path.close();


311    return path;


312  }


313


314▶   @override


315  bool shouldReclip(CustomClipper<Path> oldClipper) {


316    return true;


317  }


318}
