16/// 日期/时间滚轮选择器。


17///


18/// 与 [`TCalendar`]、[`TPicker`] 为三个独立对外组件；本组件底层复用 [`TPicker`]


19/// 滚轮能力（经内部 [`DateTimePickerWheel`]），与 [`TCalendar`] 无代码耦合。


20///


21/// 纯滚轮组件：不含工具栏、确认按钮或弹窗；选中变化通过 `onChanged` 实时回调


22///（无 `TPicker.onConfirm` 语义）。弹窗与确认请配合 `TPopup` 等自行组装。


23///


24/// `initialValue` 为非受控初始值；外部重置选中请变更 `initialValue` 或 `key`。


25/// 与 `TPicker` 不同，本组件不提供受控 `value` 参数。

103class _TDateTimePickerState extends State<TDateTimePicker> {


104  late DateTimePickerSnapshot _snapshot;


105  late int _wheelGeneration;


106


107  TDateTimePickerValue? _lastNotifiedValue;


108


109  @override


110  void initState() {


111    super.initState();


112    _wheelGeneration = 0;


113    _snapshot = _createSnapshot();


114    _lastNotifiedValue = _snapshot.toResult();


115  }


116


117  DateTimePickerSnapshot _createSnapshot() {


118    return DateTimePickerSnapshot.initial(


119      columns: widget.mode.columns,


120      initial: _resolveInitialDateTime(),


121      start: _resolveBound(widget.start),


122      end: _resolveBound(widget.end),


123      steps: widget.steps,


124    );


125  }


126


127  DateTime? _resolveBound(TDateTimePickerValue? bound) =>


128      bound?.toDateTime(fallback: kDateTimePickerDefaultFallback);


129


130  DateTime? _resolveInitialDateTime() =>


131      widget.initialValue?.toDateTime(fallback: kDateTimePickerDefaultFallback);


132


133  void _resetWheel({bool clearLastNotified = false}) {


134    if (clearLastNotified) {


135      _lastNotifiedValue = null;


136    }


137    _wheelGeneration++;


138  }


139


140  @override


141  void didUpdateWidget(covariant TDateTimePicker oldWidget) {


142    super.didUpdateWidget(oldWidget);


143    final modeChanged =


144        !_listEqualInt(oldWidget.mode.columns, widget.mode.columns);


145    final initialChanged = oldWidget.initialValue != widget.initialValue;


146    final rangeChanged =


147        oldWidget.start != widget.start || oldWidget.end != widget.end;


148    final stepsChanged = oldWidget.steps != widget.steps;


149    final showWeekChanged = oldWidget.showWeek != widget.showWeek;


150    final renderLabelChanged = oldWidget.renderLabel != widget.renderLabel;


151


152    if (!modeChanged &&


153        !initialChanged &&


154        !rangeChanged &&


155        !stepsChanged &&


156        !showWeekChanged &&


157        !renderLabelChanged) {


158      return;


159    }


160


161    if (modeChanged || initialChanged) {


162      _snapshot = DateTimePickerSnapshot.initial(


163        columns: widget.mode.columns,


164        initial: initialChanged ? _resolveInitialDateTime() : _snapshot.current,


165        start: _resolveBound(widget.start),


166        end: _resolveBound(widget.end),


167        steps: widget.steps,


168      );


169      _resetWheel(clearLastNotified: true);


170    } else {


171      _snapshot = _snapshot.rebuildFor(


172        columns: widget.mode.columns,


173        start: _resolveBound(widget.start),


174        end: _resolveBound(widget.end),


175        steps: widget.steps,


176      );


177      _resetWheel();


178    }


179  }


180


181  void _handleWheelChanged(


182    DateTimePickerSnapshot snapshot,


183    TDateTimePickerValue result,


184  ) {


185    // dateTimePicker: 滚动时仅同步 snapshot，不 setState；Wheel 自持 UI 状态。


186    _snapshot = snapshot;


187    if (_lastNotifiedValue != null && result == _lastNotifiedValue) {


188      return;


189    }


190    _lastNotifiedValue = result;


191    widget.onChanged?.call(result);


192  }

194  static bool _listEqualInt(List<dynamic> a, List<dynamic> b) {


195    if (identical(a, b)) {


196      return true;


197    }


198    if (a.length != b.length) {


199      return false;


200    }


201    for (var i = 0; i < a.length; i++) {


202      if (a[i] != b[i]) {


203        return false;


204      }


205    }


206    return true;


207  }

209  @override


210  Widget build(BuildContext context) {


211    final labels = DateTimePickerLabels.fromResource(context.resource);


212    final start = _resolveBound(widget.start);


213    final end = _resolveBound(widget.end);


214    return DateTimePickerWheel(


215      key: ValueKey<Object>(


216        Object.hash(


217          _wheelGeneration,


218          Object.hashAll(widget.mode.columns),


219          widget.start,


220          widget.end,


221          widget.showWeek,


222          widget.renderLabel,


223          widget.steps,


224          labels,


225          Localizations.localeOf(context),


226        ),


227      ),


228      snapshot: _snapshot,


229      labels: labels,


230      start: start,


231      end: end,


232      showWeek: widget.showWeek,


233      steps: widget.steps,


234      renderLabel: widget.renderLabel,


235      height: widget.height ?? 200,


236      itemCount: widget.itemCount ?? 5,


237      onChanged: _handleWheelChanged,


238    );


239  }
