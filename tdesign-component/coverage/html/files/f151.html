262  /// 获取Y坐标


263  double _getOffsetTop(Offset? widgetLocalToGlobal) {


264    var widgetBounds = _getWidgetBounds(widget.context);


265    var dy = widgetLocalToGlobal?.dy ?? 0;


266    var arrowSize = widget.showArrow ?? false ? widget.arrowSize : 0;


267    var contentSize = _getContentSize();


268    var popoverHeight = widget.height ??


269        (widget.padding != null ? widget.padding!.vertical : 24) +


270            (widget.height ?? contentSize.height);


271    switch (widget.placement) {


272      case TPopoverPlacement.bottomLeft:


273      case TPopoverPlacement.bottom:


274      case TPopoverPlacement.bottomRight:


275        return dy + (widgetBounds?.height ?? 0) + widget.offset;


276      case TPopoverPlacement.rightTop:


277      case TPopoverPlacement.leftTop:


278        return dy;


279      case TPopoverPlacement.rightBottom:


280      case TPopoverPlacement.leftBottom:


281        return dy - (popoverHeight - (widgetBounds?.height ?? 0));


282      case TPopoverPlacement.right:


283      case TPopoverPlacement.left:


284        return dy - (popoverHeight - (widgetBounds?.height ?? 0)) / 2;


285      default:


286        return dy - popoverHeight - widget.offset - arrowSize;


287    }


288  }

291  double _getOffsetLeft(Offset? widgetLocalToGlobal) {


292    var widgetBounds = _getWidgetBounds(widget.context);


293    var widgetWidth = widgetBounds?.width ?? 0;


294    var contentSize = _getContentSize();


295    var popoverWidth = widget.width ??


296        (widget.padding != null ? widget.padding!.horizontal : 24) +


297            contentSize.width;


298    var dx = widgetLocalToGlobal?.dx ?? 0;


299    switch (widget.placement) {


300      case TPopoverPlacement.topLeft:


301      case TPopoverPlacement.bottomLeft:


302        return dx;


303      case TPopoverPlacement.topRight:


304      case TPopoverPlacement.bottomRight:


305        return dx + widgetWidth - popoverWidth;


306      case TPopoverPlacement.rightTop:


307      case TPopoverPlacement.right:


308      case TPopoverPlacement.rightBottom:


309        return dx + widgetWidth + widget.offset;


310      case TPopoverPlacement.leftTop:


311      case TPopoverPlacement.left:


312      case TPopoverPlacement.leftBottom:


313        return dx - popoverWidth - widget.arrowSize - widget.offset;


314      default:


315        return dx - (popoverWidth - widgetWidth) / 2;


316    }

402  Widget _getContainerWidget() {


403    // 当未指定 width 且使用纯文本内容时，用 TextPainter 测量的宽度作为约束，


404    // 确保长文本能在 maxWidth 范围内换行（否则 Container 无宽度约束，Text 会单行无限延伸）


405    var effectiveWidth = widget.width;


406    if (effectiveWidth == null && widget.contentWidget == null) {


407      final textWidth = _getTextSize().width;


408      final paddingHorizontal = widget.padding != null ? widget.padding!.horizontal : 24;


409      effectiveWidth = textWidth + paddingHorizontal;


410    }


411    return Container(


412      width: effectiveWidth,


413      height: widget.height,


414      padding: widget.padding ?? const EdgeInsets.all(12),


415      decoration: BoxDecoration(


416          borderRadius: widget.radius ??


417              BorderRadius.circular(context.tTheme.radiusDefault),


418          color: _backgroundColor,


419          boxShadow: const [

457    var axis = CrossAxisAlignment.center;


458    var direction = VerticalDirection.down;

462      case TPopoverPlacement.bottom:


463      case TPopoverPlacement.bottomLeft:


464      case TPopoverPlacement.bottomRight:


465        direction = VerticalDirection.up;


466        break;


467      case TPopoverPlacement.right:


468      case TPopoverPlacement.rightTop:


469      case TPopoverPlacement.rightBottom:

472        children = [

485    /// 改变Row和Column交叉轴对齐位置，从而实现箭头位置


486    switch (widget.placement) {


487      case TPopoverPlacement.topLeft:


488      case TPopoverPlacement.bottomLeft:


489        axis = CrossAxisAlignment.start;


490        break;


491      case TPopoverPlacement.topRight:


492      case TPopoverPlacement.bottomRight:


493        axis = CrossAxisAlignment.end;


494        break;


495      case TPopoverPlacement.rightTop:


496      case TPopoverPlacement.leftTop:


497        axis = CrossAxisAlignment.start;


498        break;


499      case TPopoverPlacement.rightBottom:


500      case TPopoverPlacement.leftBottom:


501        axis = CrossAxisAlignment.end;


502        break;


503      default:


504        axis = CrossAxisAlignment.center;


505    }


506


507    /// 横向布局


508    if (widget.placement == TPopoverPlacement.right ||


509        widget.placement == TPopoverPlacement.rightTop ||


510        widget.placement == TPopoverPlacement.rightBottom ||


511        widget.placement == TPopoverPlacement.left ||


512        widget.placement == TPopoverPlacement.leftBottom ||


513        widget.placement == TPopoverPlacement.leftTop) {


514      return Row(


515        crossAxisAlignment: axis,


516        children: children,


517      );


518    }


519


520    /// 纵向布局


521    return Column(


522      crossAxisAlignment: axis,


523      verticalDirection: direction,


524      children: children,


525    );


526  }
