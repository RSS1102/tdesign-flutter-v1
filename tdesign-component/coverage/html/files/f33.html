103  Widget build(BuildContext context) {


104    // 获取 Theme


105    final theme = Theme.of(context).extension<TButtonThemeData>();


106    final effectiveVariant = widget.variant ?? theme?.defaultVariant ?? TButtonVariant.fill;


107    final effectiveSize = widget.size ?? theme?.defaultSize ?? TButtonSize.medium;


108    final hasGradient = theme?.gradient != null;


109


110    // 解析 ButtonStyle


111    final resolvedStyle = TButtonResolve.resolve(


112      variant: effectiveVariant,


113      colorScheme: widget.colorScheme,


114      size: effectiveSize,


115      icon: widget.icon,


116      iconPosition: widget.iconPosition,


117      theme: theme,


118      instanceStyle: widget.style,


119      context: context,


120      hasGradient: hasGradient,


121    );


122


123    // 构建带图标的内容


124    final hasIcon = widget.icon != null;


125    final hasChild = widget.child != null;


126    final iconSpacing = theme?.iconSpacing ?? 8.0;


127    final gradient = theme?.gradient;


128


129    Widget? content;


130    if (hasChild || hasIcon) {


131      final children = <Widget>[];


132


133      // 左侧图标


134      if (hasIcon && widget.iconPosition == TButtonIconPosition.left) {


135        children.add(_wrapIcon(widget.icon!, effectiveSize));


136      }


137


138      // 内容


139      if (hasChild) {


140        children.add(Flexible(child: widget.child!));


141      }


142


143      // 右侧图标


144      if (hasIcon && widget.iconPosition == TButtonIconPosition.right) {


145        children.add(_wrapIcon(widget.icon!, effectiveSize));


146      }


147


148      // 图标与文案间距


149      if (children.length == 2) {


150        children.insert(1, SizedBox(width: iconSpacing));


151      }


152


153      content = Row(


154        mainAxisSize: MainAxisSize.min,


155        mainAxisAlignment: MainAxisAlignment.center,


156        children: children,


157      );


158    }


159


160    // 使用 ElevatedButton 作为底层（Flutter 3.16+ MaterialButton 已弃用）


161    // 渐变模式：放弃 ElevatedButton，使用原生控件替代。


162    //   Web 上 ElevatedButton(MaterialType.button) 即使设置 backgroundColor=Colors.transparent


163    //   也会绘制不透明覆盖层，无法穿透显示底层渐变。


164    Widget button;


165


166    if (gradient != null) {


167      // 渐变按钮：Material(type: transparency) + InkWell + Container(gradient)


168      final shape = theme?.effectiveShape ?? TButtonShape.rectangle;


169      final borderRadius = BorderRadius.all(Radius.circular(_borderRadiusForShape(shape)));


170


171      // 从 resolvedStyle 获取前景色（用于文本/图标）


172      final isDisabled = widget.onPressed == null;


173      final states = <WidgetState>{if (isDisabled) WidgetState.disabled};


174      final fgColor = resolvedStyle.foregroundColor?.resolve(states);


175


176      // 根据 size 计算 padding


177      final padding = _gradientPadding(effectiveSize, widget.icon != null, widget.child != null, shape);


178


179      var styledContent = content;


180      if (styledContent != null && fgColor != null) {


181        styledContent = IconTheme(


182          data: IconThemeData(color: fgColor),


183          child: DefaultTextStyle(


184            style: TextStyle(color: fgColor, fontSize: _fontSizeForButton(effectiveSize)),


185            child: styledContent,


186          ),


187        );


188      }


189


190      // 最小高度约束（对齐 ElevatedButton minimumSize）


191      final minHeight = _sideLengthForSize(effectiveSize);


192      final isSquareOrCircle = shape == TButtonShape.square || shape == TButtonShape.circle;


193      final onlyIcon = widget.icon != null && widget.child == null;


194      final minWidth = (isSquareOrCircle && onlyIcon) ? minHeight : null;


195


196      button = IntrinsicWidth(


197        child: ConstrainedBox(


198          constraints: BoxConstraints(


199            minWidth: minWidth ?? 0,


200            minHeight: minHeight,


201          ),


202          child: Container(


203            decoration: BoxDecoration(gradient: gradient, borderRadius: borderRadius),


204            clipBehavior: Clip.antiAlias,


205            child: Material(


206              type: MaterialType.transparency,


207              borderRadius: borderRadius,


208              child: InkWell(


209                borderRadius: borderRadius,


210                onTap: widget.onPressed,


211                child: Padding(


212                  padding: padding,


213                  child: styledContent,


214                ),


215              ),


216            ),


217          ),


218        ),


219      );


220


221      if (theme?.margin != null) {


222        button = Container(margin: theme!.margin, child: button);


223      }


224    } else {


225      button = ElevatedButton(


226        onPressed: widget.onPressed,


227        style: resolvedStyle,


228        child: content,


229      );


230


231      // margin 外包（非渐变）


232      if (theme?.margin != null) {


233▶         button = Container(margin: theme!.margin, child: button);


234      }


235    }


236


237    return button;


238  }

240  /// 包裹图标，按 TButtonSize 设置默认尺寸


241  Widget _wrapIcon(Widget iconWidget, TButtonSize effectiveSize) {


242    if (iconWidget is Icon) {


243      final icon = iconWidget;


244      final useDefaultSize = icon.size == null;


245      final useDefaultColor = icon.color == null;


246


247      if (useDefaultSize || useDefaultColor) {


248        final iconSize = _iconSizeForButton(effectiveSize);


249        return Icon(


250          icon.icon!,


251▶           size: useDefaultSize ? iconSize : icon.size,


252▶           color: useDefaultColor ? null : icon.color,


253        );


254      }


255    }


256    return iconWidget;


257  }

273  /// 根据 shape 获取渐变裁剪圆角值


274  double _borderRadiusForShape(TButtonShape shape) {


275    final tTheme = context.tTheme;


276    return switch (shape) {


277      TButtonShape.rectangle => tTheme.radiusDefault,


278▶       TButtonShape.round => tTheme.radiusRound,


279▶       TButtonShape.square || TButtonShape.filled || TButtonShape.circle => 0,


280    };


281  }

283  /// 渐变模式下根据 size 计算 padding（与 _resolveSize 对齐）


284  EdgeInsets _gradientPadding(TButtonSize size, bool hasIcon, bool hasChild, TButtonShape shape) {


285    final isSquareOrCircle = shape == TButtonShape.square || shape == TButtonShape.circle;


286    final onlyIcon = hasIcon && !hasChild;


287


288    double padH;


289    double padV;


290


291    switch (size) {


292      case TButtonSize.large:


293        padH = onlyIcon ? 12 : 20;


294        padV = onlyIcon ? 12 : 12;


295      case TButtonSize.medium:


296        padH = onlyIcon ? 10 : 16;


297        padV = onlyIcon ? 10 : 8;


298▶       case TButtonSize.small:


299        padH = onlyIcon ? 7 : 12;


300        padV = onlyIcon ? 7 : 5;


301▶       case TButtonSize.extraSmall:


302        padH = onlyIcon ? 5 : 8;


303        padV = onlyIcon ? 5 : 3;


304    }


305


306    if (isSquareOrCircle && onlyIcon) {


307▶       return EdgeInsets.all(padH);


308    }


309    return EdgeInsets.symmetric(horizontal: padH, vertical: padV);


310  }
