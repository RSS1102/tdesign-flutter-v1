23    this.style = const TSkeletonRowColStyle(),

31


32  /// 视觉高度


33  double visualHeight(BuildContext context) {


34    var rowSpacing = style.rowSpacing(context);


35    assert(rowSpacing >= 0);


36    if (rowSpacing < 0) {


37      rowSpacing = 0;


38    }


39


40    return objects


41            .map((row) =>


42                row.fold(.0, (a, b) => a > b.visualHeight ? a : b.visualHeight))


43            .fold(.0, (a, b) => a + b) +


44        rowSpacing * (objects.length - 1);


45  }


46}
