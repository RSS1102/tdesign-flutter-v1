17  ///


18  /// [dateMode] 日期段粒度（年 / 年月 / 年月日）；不传则不展示日期列。

77        day != null &&

79        minute != null &&


80        second != null) {


81      return DateTime(year!, month!, day!, hour!, minute!, second!);


82    }


83    if (fallback == null) {


84      throw ArgumentError(


85        'TDateTimePickerValue 为 partial 值，调用 toDateTime 须显式传入 fallback',


86      );


87    }


88    return DateTime(


89      year ?? fallback.year,


90      month ?? fallback.month,


91      day ?? fallback.day,


92      hour ?? fallback.hour,


93      minute ?? fallback.minute,
