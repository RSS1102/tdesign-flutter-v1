11/// 宽高屏幕自动适配，使用举例：100.as


12extension TIntScale on int {


13  double get scale {


14    /// 屏幕的逻辑宽度=屏幕物理宽度/屏幕的像素倍数


15    var view = PlatformDispatcher.instance.views.first;


16    var screenWidth = view.physicalSize.width / view.devicePixelRatio;


17


18    /// 设计稿用的是750的宽度


19    return screenWidth / 375.0 * this;


20  }


21}
