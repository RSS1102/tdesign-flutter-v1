135class TDialogInfoWidget extends StatelessWidget {


136  const TDialogInfoWidget({


137    Key? key,


138    this.title,


139    this.titleColor,


140    this.titleAlignment,


141    this.contentWidget,


142    this.content,


143    this.contentColor,


144    this.contentMaxHeight = 0,


145    this.padding = const EdgeInsets.fromLTRB(24, 32, 24, 0),


146  }) : super(key: key);


147


148  /// 标题


149  final String? title;


150


151  /// 标题颜色


152  final Color? titleColor;


153


154  /// 标题对齐模式


155  final AlignmentGeometry? titleAlignment;


156


157  /// 内容Widget


158  final Widget? contentWidget;


159


160  /// 内容


161  final String? content;


162


163  /// 内容颜色


164  final Color? contentColor;


165


166  /// 内容的最大高度，默认为0，也就是不限制高度


167  final double contentMaxHeight;


168


169  /// 内容的内边距


170  final EdgeInsetsGeometry? padding;


171


172  @override


173  Widget build(BuildContext context) {


174    // 标题和内容不能同时为空


175    assert((title != null || content != null || contentWidget != null));


176    return Container(


177      padding: padding,


178      child: Column(


179        mainAxisSize: MainAxisSize.min,


180        children: [


181          if (title != null)


182            Align(


183              alignment: titleAlignment ?? Alignment.center,


184              child: TDialogTitle(


185                title: title,


186                titleColor: titleColor,


187              ),


188            ),


189          if (contentWidget != null || content != null)


190            Container(


191              padding: EdgeInsets.fromLTRB(


192                  0, (title != null && content != null) ? 8.0 : 0, 0, 0),


193              constraints: contentMaxHeight > 0


194                  ? BoxConstraints(


195                      maxHeight: contentMaxHeight,


196                    )


197                  : null,


198              child: contentWidget ??


199                  Scrollbar(


200                    child: SingleChildScrollView(


201                      scrollDirection: Axis.vertical,


202                      child: TDialogContent(


203                        content: content!,


204                        contentColor: contentColor ??


205                            context.tTheme.textColorSecondary,


206                      ),


207                    ),


208                  ),


209            ),


210        ],


211      ),


212    );


213  }


214}

356/// 弹窗标题


357class TDialogButton extends StatelessWidget {


358  const TDialogButton({


359    Key? key,


360    this.buttonText,


361    this.buttonTextColor,


362    this.buttonTextSize,


363    this.buttonTextFontWeight = FontWeight.w600,


364    this.buttonStyle,


365    this.buttonVariant,


366    this.buttonColorScheme,


367    required this.onPressed,


368    this.height = 40.0,


369    this.width,


370    this.isBlock = true,


371  }) : super(key: key);


372


373  /// 按钮文字


374  final String? buttonText;


375


376  /// 按钮文字颜色


377  final Color? buttonTextColor;


378


379  /// 按钮文字大小


380  final double? buttonTextSize;


381


382  /// 按钮文字粗细


383  final FontWeight? buttonTextFontWeight;


384


385  /// 按钮样式（P0 逃逸舱）


386  final ButtonStyle? buttonStyle;


387


388  /// 按钮变体类型


389  final TButtonVariant? buttonVariant;


390


391  /// 按钮配色方案


392  final TButtonColorScheme? buttonColorScheme;


393


394  /// 按钮宽度


395  final double? width;


396


397  /// 按钮高度


398  final double? height;


399


400  /// 是否通栏


401  final bool isBlock;


402


403  /// 点击回调


404  final Function() onPressed;


405


406  @override


407  Widget build(BuildContext context) {


408    final button = TButton(


409      onPressed: onPressed,


410      style: buttonStyle,


411      variant: buttonVariant ?? TButtonVariant.fill,


412      colorScheme: buttonColorScheme,


413      child: Text(


414        buttonText ?? '',


415        style: TextStyle(


416          fontWeight: buttonTextFontWeight,


417          color: buttonTextColor,


418          fontSize: buttonTextSize,


419        ),


420      ),


421    );


422


423    // 通栏布局或自定义尺寸


424    if (isBlock || width != null || height != null) {


425      return SizedBox(


426        width: isBlock ? double.infinity : width,


427        height: height,


428        child: button,


429      );


430    }


431    return button;


432  }
