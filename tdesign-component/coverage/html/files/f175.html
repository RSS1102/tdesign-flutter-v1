29class TSkeleton extends StatefulWidget {


30  factory TSkeleton({


31    Key? key,


32    TSkeletonAnimation? animation,


33    int delay = 0,


34    TSkeletonVariant variant = TSkeletonVariant.text,


35  }) {


36    assert(delay >= 0);


37


38    var objects = <List<TSkeletonRowColObj>>[];


39


40    // 根据风格创建骨架图


41    switch (variant) {


42      case TSkeletonVariant.avatar:


43        objects = const [


44          [TSkeletonRowColObj.circle()]


45        ];


46        break;


47      case TSkeletonVariant.image:


48        objects = const [


49          [


50            TSkeletonRowColObj.rect(


51              width: 72,


52              height: 72,


53              flex: null,


54            )


55          ]


56        ];


57        break;


58      case TSkeletonVariant.text:


59        objects = const [


60          [


61            TSkeletonRowColObj.text(flex: 24),


62            TSkeletonRowColObj.spacer(width: 16),


63            TSkeletonRowColObj.text(flex: 76),


64          ],


65          [TSkeletonRowColObj.text()],


66        ];


67        break;


68      case TSkeletonVariant.paragraph:


69        objects = [


70          for (int i = 0; i < 3; i++) [const TSkeletonRowColObj.text()],


71          const [


72            TSkeletonRowColObj.text(flex: 55),


73            TSkeletonRowColObj.spacer(flex: 45),


74          ],


75        ];


76        break;


77    }


78


79    return TSkeleton.fromRowCol(


80      key: key,


81      animation: animation,


82      delay: delay,


83      rowCol: TSkeletonRowCol(objects: objects),


84    );


85  }

108class _TSkeletonState extends State<TSkeleton>


109    with SingleTickerProviderStateMixin {


110  /// 动画控制器


111  late final AnimationController? _controller;


112


113  /// 动画效果


114  late final Animation<double>? _animation;


115


116  /// 加载状态


117  bool _isLoading = true;


118


119  /// 加载控件


120  static final _loadingWidget = Container();


121


122  /// 闪烁透明度


123  static const _animationFlashed = .3;


124


125  /// 静态渐变


126  static LinearGradient _animationGradient(BuildContext context) =>


127      LinearGradient(


128        colors: [


129          Colors.transparent,


130          context.tTheme.bgColorSecondaryContainerActive,


131          Colors.transparent,


132        ],


133        // 15 deg


134        begin: const Alignment(-1, -0.268),


135        end: const Alignment(1, 0.268),


136        tileMode: TileMode.clamp,


137      );


138


139  @override


140  void initState() {


141    super.initState();


142


143    // 根据动画效果创建动画控制器


144    switch (widget.animation) {


145      case TSkeletonAnimation.gradient:


146        _controller = AnimationController(


147          duration: const Duration(milliseconds: 1500),


148          vsync: this,


149        )..repeat();


150        _animation = Tween<double>(begin: -1, end: 1).animate(_controller!)


151          ..addListener(() => setState(() {}));


152        break;


153      case TSkeletonAnimation.flashed:


154        _controller = AnimationController(


155          duration: const Duration(seconds: 1),


156          vsync: this,


157        )..repeat(reverse: true);


158        _animation = Tween<double>(begin: 1, end: _animationFlashed)


159            .animate(_controller!)


160          ..addListener(() => setState(() {}));


161        break;


162      default:


163        _controller = null;


164        _animation = null;


165    }


166


167    // 延迟显示加载效果


168    Future.delayed(Duration(milliseconds: widget.delay),


169        () => setState(() => _isLoading = false));


170  }


171


172  Widget Function(TSkeletonRowColObj) _buildObj(BuildContext context) =>


173      (TSkeletonRowColObj obj) {


174        // 骨架图对象


175        Widget skeletonObj = Container(


176          width: obj.width,


177          height: obj.height,


178          margin: obj.margin,


179          decoration: BoxDecoration(


180            color: obj.style.background(context),


181            borderRadius:


182                BorderRadius.circular(obj.style.borderRadius(context)),


183          ),


184        );


185


186        // 动画效果


187        switch (widget.animation) {


188          case TSkeletonAnimation.gradient:


189            skeletonObj = ShaderMask(


190              blendMode: BlendMode.srcATop,


191              shaderCallback: (bounds) =>


192                  _animationGradient(context).createShader(


193                Rect.fromLTWH(


194                  bounds.width * _animation!.value,


195                  0,


196                  bounds.width,


197                  bounds.height,


198                ),


199              ),


200              child: skeletonObj,


201            );


202            break;


203          case TSkeletonAnimation.flashed:


204            skeletonObj = Opacity(


205              opacity: _animation!.value,


206              child: skeletonObj,


207            );


208            break;


209          default:


210        }


211


212        // 根据弹性因子创建弹性布局


213        return obj.flex == null


214            ? skeletonObj


215            : Flexible(flex: obj.flex!, child: skeletonObj);


216      };


217


218  @override


219  Widget build(BuildContext context) {


220    // 加载状态返回空容器


221    if (_isLoading) {


222      return _loadingWidget;


223    }


224


225    if (widget.rowCol.objects.length == 1) {


226      return widget.rowCol.objects.first.length == 1


227          // 单个对象


228          ? _buildObj(context)(widget.rowCol.objects.first.first)


229          // 单行多个对象


230          : Flexible(


231              child: Row(


232              children:


233                  widget.rowCol.objects.first.map(_buildObj(context)).toList(),


234            ));


235    }


236


237    // 多行多个对象


238    List<Widget> skeletonRows = widget.rowCol.objects


239        .map((row) => Row(children: row.map(_buildObj(context)).toList()))


240        .toList();


241    final rowSpacing = widget.rowCol.style.rowSpacing(context);


242    if (rowSpacing > 0) {


243      skeletonRows = skeletonRows


244          .expand((row) => [row, SizedBox(height: rowSpacing)])


245          .toList();


246      if (skeletonRows.isNotEmpty) {


247        skeletonRows.removeLast();


248      }


249    } // 添加行间距


250    var skeletonRowCol = Column(children: skeletonRows); // 行列布局


251


252    return widget.rowCol.objects


253            .any((row) => row.any((obj) => obj.flex != null))


254        // 添加弹性布局


255        ? Flexible(


256            child: Container(


257              constraints: BoxConstraints(


258                maxHeight: widget.rowCol.visualHeight(context),


259              ), // 限制最大高度


260              child: skeletonRowCol,


261            ),


262          )


263        : skeletonRowCol;


264  }

266  @override


267  void dispose() {

269    super.dispose();
