90  // 按插入顺序比较联动树（展示顺序即插入顺序）


91  static bool _treeEqual(


92      Map<TPickerOption, dynamic> a, Map<TPickerOption, dynamic> b) {


93    if (identical(a, b)) {


94      return true;


95    }


96    if (a.length != b.length) {


97      return false;


98    }


99    final aEntries = a.entries.iterator;


100    final bEntries = b.entries.iterator;


101    while (aEntries.moveNext() && bEntries.moveNext()) {


102      if (aEntries.current.key != bEntries.current.key ||


103          !_childEqual(aEntries.current.value, bEntries.current.value)) {


104        return false;


105      }


106    }


107    return true;


108  }

111    if (identical(a, b)) {


112      return true;


113    }


114▶     if (a is Map<TPickerOption, dynamic> && b is Map<TPickerOption, dynamic>) {

118▶       return listEquals(a, b);
