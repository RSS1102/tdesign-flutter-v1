246class _TTabBarIndicatorPainter extends BoxPainter {


247  static const double _defaultIndicatorWidth = 16;


248  static const double _defaultIndicatorHeight = 3;


249


250  final TTabBarIndicator decoration;


251  final _paint = Paint();


252


253  _TTabBarIndicatorPainter(this.decoration, VoidCallback onChanged) {


254    _paint.color = decoration.indicatorColor ??


255        decoration.context!.tTheme.brandNormalColor;


256    _paint.strokeCap = StrokeCap.round;


257  }


258


259  @override


260  void paint(Canvas canvas, Offset offset, ImageConfiguration configuration) {


261    canvas.drawLine(


262        Offset(offset.dx + (configuration.size!.width - _indicatorWidth()) / 2,


263            configuration.size!.height - _indicatorHeight() / 2),


264        Offset(offset.dx + (configuration.size!.width + _indicatorWidth()) / 2,


265            configuration.size!.height - _indicatorHeight() / 2),


266        _paint..strokeWidth = _indicatorHeight());


267  }


268


269  double _indicatorHeight() =>


270      decoration.indicatorHeight ?? _defaultIndicatorHeight;


271


272  double _indicatorWidth() =>


273      decoration.indicatorWidth ?? _defaultIndicatorWidth;


274}
