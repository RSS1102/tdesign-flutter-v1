233  CrossAxisAlignment _getAlign(TCellAlign align) {


234    switch (align) {


235      case TCellAlign.top:


236        return CrossAxisAlignment.start;


237      case TCellAlign.middle:


238        return CrossAxisAlignment.center;


239      case TCellAlign.bottom:


240        return CrossAxisAlignment.end;

244  void _setStatus(String status, int milliseconds, bool hover) {


245    if (_disabled || !hover) {


246      return;


247    }


248    if (milliseconds == 0) {


249      setState(() {


250        _status = status;


251      });


252      return;


253    }


254    Future.delayed(Duration(milliseconds: milliseconds), () {


255      setState(() {


256        _status = status;


257      });


258    });


259  }
