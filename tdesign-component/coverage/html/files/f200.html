480  Widget _buildAnimatedIndicator(BuildContext context, double itemWidth) {


481    // 只有 label 样式才显示背景指示器


482    if (widget.componentType != TBottomTabBarComponentType.label) {


483      return const SizedBox.shrink();


484    }


485


486    // 无动画模式不显示（由各个 item 自己渲染）


487    if (widget.indicatorAnimation == TBottomTabBarIndicatorAnimation.none) {


488      return const SizedBox.shrink();


489    }


490


491    final animValue = _animation?.value ?? _selectedIndex.toDouble();


492


493    switch (widget.indicatorAnimation) {


494      case TBottomTabBarIndicatorAnimation.linear:


495        return _buildLinearIndicator(context, itemWidth, animValue);


496      case TBottomTabBarIndicatorAnimation.elastic:


497        return _buildElasticIndicator(context, itemWidth, animValue);


498▶       case TBottomTabBarIndicatorAnimation.none:


499        return const SizedBox.shrink();


500    }


501  }

504  Widget _buildLinearIndicator(

506    final horizontalPadding = widget.navigationTabs.length > 3 ? 8.0 : 12.0;

514▶         widget.basicType == TBottomTabBarBasicType.expansionPanel

531  /// 弹性拉伸指示器


532  Widget _buildElasticIndicator(


533      BuildContext context, double itemWidth, double animValue) {


534    final horizontalPadding = widget.navigationTabs.length > 3 ? 8.0 : 12.0;


535


536    // 计算起始和目标索引


537    final fromIndex = animValue.floor();


538    final toIndex = animValue.ceil();


539    final progress = animValue - fromIndex;


540


541    // 弹性曲线：前半段快速拉伸，后半段缓慢收缩


542    double width;


543    double left;


544


545    if (progress < 0.5) {


546      // 前半段：从起点向终点拉伸


547      final stretchProgress = progress * 2; // 0 -> 1


548      width = (itemWidth - horizontalPadding * 2) *


549          (1 + stretchProgress * (toIndex - fromIndex));


550      left = fromIndex * itemWidth + horizontalPadding;


551    } else {


552      // 后半段：从终点收缩到正常宽度


553▶       final shrinkProgress = (progress - 0.5) * 2; // 0 -> 1


554▶       width = (itemWidth - horizontalPadding * 2) *


555▶           (1 + (1 - shrinkProgress) * (toIndex - fromIndex));


556▶       left = fromIndex * itemWidth +


557▶           horizontalPadding +


558▶           shrinkProgress * (toIndex - fromIndex) * itemWidth;


559    }


560


561    // 计算高度


562    final height = widget.basicType == TBottomTabBarBasicType.text ||


563▶         widget.basicType == TBottomTabBarBasicType.expansionPanel


564        ? 32.0


565        : null;


566


567    return Positioned(


568      left: left,


569      child: Container(


570        width: width,


571        height: height,


572        decoration: BoxDecoration(


573          color: _effectiveSelectedBgColor,


574          borderRadius: const BorderRadius.all(Radius.circular(24)),


575        ),


576      ),


577    );


578  }

580  Widget _item(int index, double itemWidth) {


581    var tabItemConfig = widget.navigationTabs[index];


582    // iconText 且存在 centerDistance 间距时，压缩上下内边距为图标+文本+间距腾出空间，


583    // 避免 Column 内容溢出（centerDistance 默认为 0，不影响常规渲染与 Golden 基线）。


584    final isIconTextWithGap = widget.basicType ==


585            TBottomTabBarBasicType.iconText &&


586        _effectiveCenterDistance > 0;


587    return Container(


588        height: _effectiveBarHeight,


589        width: itemWidth,


590        alignment: Alignment.center,


591        padding: EdgeInsets.only(


592            top: isIconTextWithGap ? 4 : 7,


593            bottom: isIconTextWithGap


594                ? 1


595                : (widget.basicType == TBottomTabBarBasicType.iconText ? 5 : 7)),


596        child: TBottomTabBarItemWithBadge(


597          basicType: widget.basicType,


598          componentType:


599              widget.componentType ?? TBottomTabBarComponentType.label,


600          outlineType: widget.outlineType ?? TBottomTabBarOutlineType.filled,


601          itemConfig: tabItemConfig,


602          isSelected: index == _selectedIndex,


603          itemHeight: _effectiveBarHeight,


604          itemWidth: itemWidth,


605          tabsLength: widget.navigationTabs.length,


606          selectedBgColor: _effectiveSelectedBgColor,


607          unselectedBgColor: _effectiveUnselectedBgColor,


608          centerDistance: _effectiveCenterDistance,


609          needInkWell: _effectiveNeedInkWell,


610          showItemBackground:


611          widget.indicatorAnimation == TBottomTabBarIndicatorAnimation.none,


612          onTap: () {


613            _onTap(index);


614          },


615          onLongPress: () {


616            tabItemConfig.onLongPress?.call();


617          },


618        ));


619  }

759  Widget _constructItem(


760      BuildContext context, BadgeConfig? badgeConfig, bool isInOrOutCapsule) {

791      child = IconTheme(

795              : context.tTheme.textColorPrimary,

799    }

803      var unSelectedIcon = itemConfig.unselectedIcon;


804      child = Column(


805        mainAxisAlignment: MainAxisAlignment.center,


806        crossAxisAlignment: CrossAxisAlignment.center,


807        children: [


808          IconTheme(


809            data: IconThemeData(


810              color: isSelected


811                  ? context.tTheme.brandNormalColor


812                  : context.tTheme.textColorPrimary,


813            ),


814            child: isSelected ? selectedIcon! : unSelectedIcon!,


815          ),


816          if (centerDistance > 0)


817            SizedBox(


818              height: centerDistance,


819            ),

821              ? _textItem(


822                  context,


823                  itemConfig,


824                  isSelected,


825                  context.tTheme.fontBodyExtraSmall!,


826                )


827              : Container()


828        ],


829      );


830    }


831


832    var top = badgeConfig?.badgeTopOffset ?? -2;


833    var right = badgeConfig?.badgeRightOffset ?? -10;


834    return Stack(


835      clipBehavior: Clip.none,

846  Widget _textItem(BuildContext context, TBottomTabBarTabConfig config,


847      bool isSelected, Font font) {


848    return TText(


849      config.tabText,


850      font: font,


851      fontWeight: isSelected ? FontWeight.bold : FontWeight.normal,


852      style:


853          isSelected ? config.selectTabTextStyle : config.unselectTabTextStyle,


854      textColor: isSelected


855          ? context.tTheme.brandNormalColor


856          : context.tTheme.textColorPrimary,


857      forceVerticalCenter: true,


858    );


859  }

861  _buildItem(BuildContext context) {


862    var badgeConfig = itemConfig.badgeConfig;


863    var isInOrOutCapsule = componentType == TBottomTabBarComponentType.label ||


864        outlineType == TBottomTabBarOutlineType.capsule;


865


866    // centerDistance > 0 时进一步压缩顶部内边距，为图标与文本的间距腾出空间


867    final reduceTopPad =


868        basicType == TBottomTabBarBasicType.iconText && centerDistance > 0;


869    var child = Container(


870      alignment: Alignment.center,


871      padding: EdgeInsets.only(


872        top: (isInOrOutCapsule ? 3.0 : 2.0) - (reduceTopPad ? 1.0 : 0.0),


873        bottom: isInOrOutCapsule


874            ? (basicType == TBottomTabBarBasicType.iconText ? 0.0 : 1.0)


875            : 0.0,


876      ),


877      color: Colors.transparent,


878      child: _constructItem(context, badgeConfig, isInOrOutCapsule),


879    );


880


881    if (!needInkWell) {


882      return child;


883    }


884    return Material(


885      color: Colors.transparent,


886      borderRadius: isInOrOutCapsule ? BorderRadius.circular(24) : null,


887      child: InkWell(


888        borderRadius: isInOrOutCapsule ? BorderRadius.circular(24) : null,


889        splashFactory: InkRipple.splashFactory,


890        splashColor: selectedBgColor ?? context.tTheme.brandLightColor,


891        highlightColor: selectedBgColor ?? context.tTheme.brandLightColor,


892▶         onTap: () => handleTap(context),


893        child: child,


894      ),


895    );


896  }


897


898  void handleTap(BuildContext context) {


899    onTap.call();


900


901    var popUpButtonConfig = itemConfig.popUpButtonConfig;


902    if (popUpButtonConfig != null) {


903      Navigator.push(


904          context,


905          PopRoute(


906            child: PopupDialog(


907              itemWidth - _kDefaultMenuItemWidthShrink,


908              btnContext: context,


909              config: popUpButtonConfig.popUpDialogConfig,


910              items: popUpButtonConfig.items,


911              onClickMenu: (value) {


912                popUpButtonConfig.onChanged(value);


913              },


914            ),


915          ));


916    }


917  }

1074  @override

1077    button = widget.btnContext.findRenderObject() as RenderBox;


1078    size = button!.size;


1079    overlay =


1080        Overlay.of(widget.btnContext).context.findRenderObject() as RenderBox;


1081    position = RelativeRect.fromRect(


1082      Rect.fromPoints(


1083        button!.localToGlobal(Offset.zero, ancestor: overlay),

1090  @override

1092    var popUpItemHeight =


1093        widget.config?.popUpItemHeight ?? _kDefaultMenuItemHeight;


1094    var popUpItemWidth = widget.config?.popUpWidth ?? widget.defaultPopUpWidth;


1095    var menuItems = widget.items


1096        .map((e) => GestureDetector(


1097            behavior: HitTestBehavior.opaque,


1098            onTap: () {


1099              widget.onClickMenu(e.value);


1100              Navigator.of(context).pop();


1101            },


1102            child: SizedBox(

1109    final popUpPanelHeight = popUpItemHeight * widget.items.length +


1110        (widget.config?.arrowHeight ?? _kArrowHeight);


1111    final screenHeight = MediaQuery.of(context).size.height;

1117    final safeTop = rawTop.clamp(8.0, screenHeight - popUpPanelHeight - 8.0);


1118


1119    return Material(


1120      type: MaterialType.transparency,


1121      child: GestureDetector(


1122▶         onTap: () => Navigator.of(context).pop(),

1131                top: safeTop,


1132                right: position!.right - (popUpItemWidth + size!.width) / 2,

1134                  width: popUpItemWidth,


1135                  height: popUpItemHeight * widget.items.length +


1136                      (widget.config?.arrowHeight ?? _kArrowHeight),

1138                      BoxDecoration(boxShadow: context.tTheme.shadowsTop),


1139                  child: CustomPaint(


1140                    painter: PanelWithDownArrow(

1142                        backgroundColor: widget.config?.backgroundColor ??

1145                      alignment: Alignment.topCenter,


1146                      height: popUpItemHeight * widget.items.length,

1181/// 带下箭头的展开panel


1182class PanelWithDownArrow extends CustomPainter {


1183  /// 弹出面板形状配置


1184  TBottomTabBarPopUpShapeConfig? config;


1185


1186  /// 背景颜色


1187  Color backgroundColor;


1188


1189  PanelWithDownArrow({this.config, required this.backgroundColor});


1190


1191  @override


1192  void paint(Canvas canvas, Size size) {


1193    var paint = Paint()


1194      ..isAntiAlias = true


1195      ..color = backgroundColor


1196      ..style = PaintingStyle.fill;


1197    var path = Path();


1198    var panelWidth = size.width;


1199    var panelHeight = size.height - (config?.arrowHeight ?? _kArrowHeight);


1200


1201    canvas.drawRRect(


1202        RRect.fromRectAndRadius(Rect.fromLTWH(0, 0, panelWidth, panelHeight),


1203            Radius.circular(config?.radius ?? 0.0)),


1204        paint);


1205


1206    /// 下方箭头


1207    if (config?.arrowWidth != 0.0 && config?.arrowHeight != 0.0) {


1208      var left = (panelWidth - _kArrowWidth) / 2;


1209      var right = (panelWidth + _kArrowWidth) / 2;


1210      var bottom = panelHeight + _kArrowHeight;


1211      if (config?.arrowWidth != null) {


1212▶         left = (panelWidth - config!.arrowWidth!) / 2;


1213▶         right = (panelWidth + config!.arrowWidth!) / 2;


1214      }


1215      if (config?.arrowHeight != null) {


1216▶         bottom = panelHeight + config!.arrowHeight!;


1217      }


1218


1219      path.moveTo(left, panelHeight);


1220      path.lineTo(panelWidth / 2, bottom);


1221      path.lineTo(right, panelHeight);


1222      canvas.drawPath(path, paint);


1223    }


1224  }
