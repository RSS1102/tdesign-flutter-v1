252  /// 计算文本渲染宽度


253  double _measureTextWidth(


254      String? text, TextStyle? style, BuildContext context) {


255    if (text == null || text.isEmpty) {


256      return 0;


257    }


258    final effectiveStyle = (style ?? const TextStyle()).copyWith(


259      fontSize: context.tTheme.fontBodyLarge?.size,


260      letterSpacing: 0,


261      height: 1.0,


262    );


263    final textPainter = TextPainter(


264      text: TextSpan(text: text, style: effectiveStyle),


265      textDirection: TextDirection.ltr,


266      maxLines: 1,


267    )..layout();


268    return textPainter.width + 2;


269  }

271  /// 计算输入框左侧信息总宽度


272  double _calculateLeftInfoWidth(


273      BuildContext context, TInputThemeData? theme, TInputSpacer spacer) {


274    final iconSpace = prefix != null ? (spacer.iconLabelSpace ?? 4) : 0;


275    final iconWidth = prefix != null ? 24 + iconSpace : 0;


276    final labelWidth = _measureTextWidth(label, labelStyle, context);


277    final requiredWidth = (required ?? false) ? 14 : 0;


278    return iconWidth + labelWidth + requiredWidth + (leftInfoWidth ?? 4);


279  }
