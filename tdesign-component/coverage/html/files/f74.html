132  /// 命令式刷新本列 options / controller，不重建其它列。


133  void applyColumnUpdate({


134    required List<TPickerOption> options,


135    required FixedExtentScrollController controller,


136  }) {


137    // 与外层 didUpdateWidget 重复 guard：applyColumnUpdate 还有从 picker


138    // 命令式调用的入口（不经过 didUpdateWidget），需自带防御性短路。


139    if (listEquals(_options, options) && _controller == controller) {


140      return;


141    }


142    final previousController = _controller;


143    setState(() {


144      _options = options;


145      _controller = controller;


146      // 切 controller 时清空 dedup 缓存：旧 controller 的 _lastNotifiedIndex


147      // 与新 controller 数值上重合时,_animateToNearestEnabled 的


148      // `if (_lastNotifiedIndex == target) return;` 会误判为已通知,


149      // 漏掉一次 onChange(onAnimationComplete 路径)。极端低概率但状态


150      // 本应与 controller 同寿命,一行清零根治。


151      if (!identical(previousController, controller)) {


152        _lastNotifiedIndex = null;


153      }


154    });

156      WidgetsBinding.instance.addPostFrameCallback((_) {


157        if (!identical(_controller, previousController)) {

159        }

178  /// 无障碍手势用：严格上一/下一格（与 Flutter `CupertinoPicker._handleIncrease`


179  /// 同款契约 —— 一次只移动一格，**不跨 disabled**）。


180  ///


181  /// disabled 项的修正由 [animateToNearestEnabled]（即物理滚动结束后的修正流程）


182  /// 统一处理，a11y 路径保持"按一下=移一格"的心智模型纯净。


183  /// 边界时返回 false（外层可据此关闭 increase/decrease 语义）。


184  bool nudge(int delta) {


185    if (_options.isEmpty || _isAnimating || widget.disabled) {


186      return false;


187    }


188    // H3 防御：与 currentSelectedIndex 同款，controller 在首次 build 完


189    // 成前无 client，selectedItem 会抛断言（_handleScrollEnd 早于 ListView


190    // 挂载触发时可能命中此分支）。


191    if (!_controller.hasClients) {


192      return false;


193    }


194    final current = _controller.selectedItem;


195    if (current < 0 || current >= _options.length) {


196      return false;


197    }


198    final target = current + delta;


199    if (target < 0 || target >= _options.length) {


200      return false;


201    }


202    _controller.jumpToItem(target);


203    widget.onItemSelected(widget.colIndex, target, _options);


204    return true;


205  }

207  /// 从 [start] 出发搜索最近一个 enabled 索引（双向同时推进，先命中者胜出；


208  /// 同距离偏向前向）。


209  ///


210  /// 仅供 [animateToNearestEnabled] 内部使用（disabled 项修正动画）：


211  /// a11y 的 `onIncrease` / `onDecrease` 不调用此方法（严格 ±1 步进，


212  /// disabled 修正由 controller notify 后统一处理）。


213  /// 全 disabled 或无可达项时返回 -1。


214  static int nearestEnabledIndex(List<TPickerOption> data, int start) {


215    for (var s = 1; s < data.length; s++) {


216      final forward = start + s;


217      if (forward < data.length && !data[forward].disabled) {


218        return forward;


219      }


220      final backward = start - s;


221      if (backward >= 0 && !data[backward].disabled) {


222        return backward;


223      }


224    }


225▶     return -1;


226  }

228  void _animateToNearestEnabled(int currentIndex) {


229    if (!mounted || _options.isEmpty) {


230      return;


231    }


232    final target = nearestEnabledIndex(_options, currentIndex);


233    if (target < 0 || target == currentIndex) {


234      return;


235    }


236


237    if (!_isAnimating) {


238      _isAnimating = true;


239    }


240


241    WidgetsBinding.instance.addPostFrameCallback((_) {


242      if (!mounted || !_isAnimating) {


243        return;


244      }


245      final c = _controller;


246      final newIndex = c.selectedItem;


247      // 动画调度到这一帧前，用户可能已经手动滚到 enabled 项了


248      if (newIndex >= 0 &&


249          newIndex < _options.length &&


250          !_options[newIndex].disabled) {


251▶         _isAnimating = false;


252        return;


253      }


254


255      final distance = (target - newIndex).abs();


256      final ms = distance <= kCorrectAnimDistanceThreshold


257          ? kCorrectAnimShortMs


258          : kCorrectAnimLongMs;


259      c


260          .animateToItem(


261        target,


262        duration: Duration(milliseconds: ms),


263        curve: Curves.easeOutCubic,


264      )


265          .then((_) {


266        if (!mounted) {


267          return;


268        }


269        _isAnimating = false;


270        // H2 dedup：动画过程中 `onSelectedItemChanged` 已经在落点触发过


271        // `onItemSelected`，这里再调 `onAnimationComplete` 会让父组件


272        // 重复收到同 index 的 onChange。命中 dedup 时跳过。


273        if (_lastNotifiedIndex == target) {


274          return;


275        }


276▶         widget.onAnimationComplete?.call(widget.colIndex, target, _options);


277      }).catchError((Object e, StackTrace stack) {


278▶         debugPrint('WheelColumn animation interrupted: $e');


279▶         _isAnimating = false;


280      });


281    });


282  }

284  /// 滚动结束后：若停在 disabled 项上则修正到最近 enabled


285  void _handleScrollEnd() {


286    if (widget.disabled || _options.isEmpty || _isAnimating) {


287      return;


288    }


289    final currentIndex = _controller.selectedItem;


290    if (currentIndex < 0 ||


291        currentIndex >= _options.length ||


292        !_options[currentIndex].disabled) {


293      return;


294    }


295    _animateToNearestEnabled(currentIndex);


296  }
