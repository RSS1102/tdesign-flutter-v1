477    final title = widget.title;


478    final customContent =


479        widget.customContentBuilder ?? groupState?.widget.customContentBuilder;

481    var content = customContent?.call(context, checked, title);


482    if (content == null) {


483      if (title != null || customContent != null && title != null) {


484        content = TText(title,


485            maxLines: widget.titleMaxLine ?? groupState?.widget.titleMaxLine,


486            overflow: TextOverflow.ellipsis,


487            textColor: widget.enabled

491                context.tTheme


492                    .fontBodyLarge); // TODO custom fontSize https://github.com/Tencent/tdesign-flutter/issues/66


493      }


494    }


495    return content;


496  }

550class RadioCorner extends CustomPainter {


551  final double length;


552  final double radius;


553  final Color fillColor;


554


555  RadioCorner(


556      {required this.length, required this.radius, required this.fillColor});


557


558  @override


559  void paint(Canvas canvas, Size size) {


560    var paint = Paint()


561      ..isAntiAlias = true


562      ..strokeWidth = 1


563      ..color = fillColor


564      ..style = PaintingStyle.fill;


565    var rect = Rect.fromCircle(center: Offset(radius, radius), radius: radius);


566    var pi = 3.1415;


567    var path = Path();


568    path.moveTo(0, radius);


569    path.addArc(


570      rect,


571      180 * (pi / 180.0),


572      90 * (pi / 180.0),


573    );


574    path.moveTo(radius, 0);


575    path.lineTo(length, 0);


576    path.lineTo(0, length);


577    path.lineTo(0, radius);


578    canvas.drawPath(path, paint);


579  }
