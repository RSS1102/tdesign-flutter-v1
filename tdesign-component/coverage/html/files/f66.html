23/// `TDateTimePicker` 各列默认 label 的文案配置（@internal）。


24///


25/// build 时由 `fromResource` 从 `TResourceDelegate` 生成；缺省文案见


26/// `TResourceManager.defaultDelegate`（`_DefaultResourceDelegate`）。


27@internal


28@immutable


29class DateTimePickerLabels {


30  const DateTimePickerLabels({


31    required this.unitSuffix,


32    required this.weekLabels,


33  });


34


35  /// 未注入 `TResourceManager.setResourceBuilder` 时的 label（派生自 `_DefaultResourceDelegate`）。


36  static final DateTimePickerLabels defaults =


37      DateTimePickerLabels.fromResource(TResourceManager.defaultDelegate);


38


39  final Map<DateTimeColumn, String> unitSuffix;


40  final List<String> weekLabels;


41


42  factory DateTimePickerLabels.fromResource(TResourceDelegate resource) {


43    return DateTimePickerLabels(


44      unitSuffix: {


45        DateTimeColumn.year: resource.yearLabel,


46        DateTimeColumn.month: resource.monthLabel,


47        DateTimeColumn.day: resource.dateLabel,


48        DateTimeColumn.hour: resource.hours,


49        DateTimeColumn.minute: resource.minutes,


50        DateTimeColumn.second: resource.seconds,


51      },


52      weekLabels: _weekLabelsFromResource(resource),


53    );


54  }


55


56  String formatColumn(DateTimeColumn column, int value) =>


57      '$value${unitSuffix[column]}';


58


59  String weekdayLabel(int weekday) => weekLabels[weekday - 1];


60


61  static List<String> _weekLabelsFromResource(TResourceDelegate resource) {


62    String label(String shortName) {


63      if (shortName.length == 1 && resource.weeksLabel.isNotEmpty) {


64        return '${resource.weeksLabel}$shortName';


65      }


66      return shortName;


67    }


68


69    return [


70      label(resource.monday),


71      label(resource.tuesday),


72      label(resource.wednesday),


73      label(resource.thursday),


74      label(resource.friday),


75      label(resource.saturday),


76      label(resource.sunday),


77    ];


78  }


79


80  @override


81  bool operator ==(Object other) =>


82      identical(this, other) ||


83      other is DateTimePickerLabels &&


84          mapEquals(unitSuffix, other.unitSuffix) &&


85          listEquals(weekLabels, other.weekLabels);


86


87  @override


88  int get hashCode => Object.hash(


89      Object.hashAll(unitSuffix.entries), Object.hashAll(weekLabels));


90}

118/// 把日期粒度 + 时间粒度展开成 `DateTimeColumn` 序列。


119List<DateTimeColumn> _expand(DateMode? date, TimeMode? time) {


120  final cols = <DateTimeColumn>[];


121  switch (date) {


122    case DateMode.year:


123      cols.add(DateTimeColumn.year);


124    case DateMode.month:


125      cols


126        ..add(DateTimeColumn.year)


127        ..add(DateTimeColumn.month);


128    case DateMode.date:


129      cols


130        ..add(DateTimeColumn.year)


131        ..add(DateTimeColumn.month)


132        ..add(DateTimeColumn.day);


133    case null:


134      break;


135  }


136  switch (time) {


137    case TimeMode.hour:


138      cols.add(DateTimeColumn.hour);


139    case TimeMode.minute:


140      cols


141        ..add(DateTimeColumn.hour)


142        ..add(DateTimeColumn.minute);


143    case TimeMode.second:


144      cols


145        ..add(DateTimeColumn.hour)


146        ..add(DateTimeColumn.minute)


147        ..add(DateTimeColumn.second);


148    case null:


149      break;


150  }


151  assert(


152    cols.isNotEmpty,


153    'DateTimePickerMode: dateMode 与 timeMode 不能同时为 null',


154  );


155  return List<DateTimeColumn>.unmodifiable(cols);


156}
