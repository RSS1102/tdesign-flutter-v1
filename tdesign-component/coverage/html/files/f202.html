822class ChevronPainter extends CustomPainter {


823  ChevronPainter({


824    required this.upColor,


825    required this.downColor,


826  });


827


828  /// 线条颜色（向上）


829  final Color upColor;


830


831  /// 线条颜色（向下）


832  final Color downColor;


833


834  @override


835  void paint(Canvas canvas, Size size) {


836    final upPaint = Paint()


837      ..color = upColor


838      ..strokeWidth = 1.4


839      ..style = PaintingStyle.stroke


840      ..strokeCap = StrokeCap.round;


841


842    final clientX = size.width;


843    final clientY = size.height;


844    final centerX = clientX / 2;


845    final centerY = clientY / 2;


846


847    // 向上箭头


848    final upPath = Path();


849    upPath.moveTo(3.6, centerY - 1.8);


850    upPath.lineTo(centerX, 2);


851    upPath.lineTo(clientX - 3.6, centerY - 1.8);


852    upPath.close();


853


854    // 向下箭头


855    final downPaint = Paint()


856      ..color = downColor


857      ..strokeWidth = 1.5


858      ..style = PaintingStyle.stroke


859      ..strokeCap = StrokeCap.round;


860    final downPath = Path();


861    downPath.moveTo(3.6, centerY + 1.8);


862    downPath.lineTo(centerX, clientY - 2);


863    downPath.lineTo(clientX - 3.6, centerY + 1.8);


864    downPath.close();


865


866    canvas.drawPath(upPath, upPaint);


867    canvas.drawPath(downPath, downPaint);


868  }


869


870▶   @override


871  bool shouldRepaint(covariant CustomPainter oldDelegate) {


872    return true;


873  }
