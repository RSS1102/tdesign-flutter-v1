12/// 分割线组件


13///


14/// T3 自绘层级，不包装 Material [Divider]。包含两种绘制模式：


15/// - 模式 A（纯线）：[child] 为空，横线可虚线、竖线强制实线


16/// - 模式 B（线 + 中间）：[layout] 为 horizontal 且 [child] 非空


17///


18/// 竖线（[TDividerLayout.vertical]）时强制忽略 [dashed]、[align]、[child]，


19/// 对齐 React/Vue Divider 跨端规范。


20///


21/// 示例：


22/// ```dart


23/// // 水平分割线


24/// TDivider()


25///


26/// // 带文字的水平分割线


27/// TDivider(child: Text('文字信息'))


28///


29/// // 虚线 + 文字靠左


30/// TDivider(dashed: true, align: TDividerAlign.left, child: Text('靠左'))


31///


32/// // 竖线（定高由父布局控制）


33/// SizedBox(height: 56, child: TDivider(layout: TDividerLayout.vertical))


34/// ```


35class TDivider extends StatelessWidget {


36  const TDivider({


37    super.key,


38    this.layout,


39    this.align,


40    this.dashed,


41    this.child,


42  });


43


44  /// 横/竖分割线，默认 [TDividerLayout.horizontal]


45  final TDividerLayout? layout;


46


47  /// 中间内容在线条中的位置，默认 [TDividerAlign.center]


48  /// 仅 [TDividerLayout.horizontal] 生效


49  final TDividerAlign? align;


50


51  /// 是否为虚线，默认 false


52  /// 仅 [TDividerLayout.horizontal] 生效


53  final bool? dashed;


54


55  /// 中间子元素（替代 0.2.x 的 text + widget 双通道）


56  /// 纯文案用 `child: Text('……')`


57  final Widget? child;


58


59  @override


60  Widget build(BuildContext context) {


61    // ---- resolve 内联链 ----


62    // 优先级：构造器 L1/L2 > TDividerThemeData > Material DividerTheme (P2) > Token (P4)


63    final theme = Theme.of(context).extension<TDividerThemeData>();


64    final dividerTheme = Theme.of(context).dividerTheme;


65


66    final effectiveLayout = layout ?? TDividerLayout.horizontal;


67


68    // L4 值按优先级 fallback


69    final effectiveColor =


70        theme?.color ?? dividerTheme.color ?? context.tTheme.componentStrokeColor;


71    final effectiveThickness = theme?.thickness ?? dividerTheme.thickness ?? 0.5;


72    final effectiveIndent = theme?.indent;


73    final effectiveEndIndent = theme?.endIndent;


74    final effectiveGapPadding = theme?.gapPadding ?? const EdgeInsets.symmetric(horizontal: 8);


75    final effectiveMargin = theme?.margin;


76    final effectiveTextStyle = theme?.textStyle;


77


78    // ---- 按 layout 选模式 ----


79    Widget result;


80


81    if (effectiveLayout == TDividerLayout.vertical) {


82      // 竖线：仅模式 A，强制忽略 dashed / align / child


83      result = _buildVerticalLine(effectiveColor, effectiveThickness);


84    } else if (child != null) {


85      // 模式 B：横线 + 中间内容


86      final effectiveDashed = dashed ?? false;


87      final effectiveAlign = align ?? TDividerAlign.center;


88      result = _buildLineWithChild(


89        effectiveColor,


90        effectiveThickness,


91        effectiveGapPadding,


92        effectiveTextStyle,


93        effectiveDashed,


94        effectiveAlign,


95      );


96    } else {


97      // 模式 A：纯横线


98      final effectiveDashed = dashed ?? false;


99      result = _buildHorizontalLine(


100        effectiveColor,


101        effectiveThickness,


102        effectiveDashed,


103        effectiveIndent,


104        effectiveEndIndent,


105      );


106    }


107


108    // 外边距


109    if (effectiveMargin != null) {


110      result = Padding(padding: effectiveMargin, child: result);


111    }


112


113    return result;


114  }


115


116  /// 模式 A — 纯横线（实线或虚线）


117  Widget _buildHorizontalLine(


118    Color color,


119    double thickness,


120    bool isDashed,


121    double? indent,


122    double? endIndent,


123  ) {


124    Widget line;


125    if (isDashed) {


126      line = SizedBox(


127        height: thickness,


128        child: CustomPaint(


129          painter: DashedPainter(color: color, strokeWidth: thickness),


130        ),


131      );


132    } else {


133      line = Container(height: thickness, color: color);


134    }


135


136    // 缩进


137    if (indent != null || endIndent != null) {


138      line = Padding(


139        padding: EdgeInsetsDirectional.only(


140          start: indent ?? 0,


141          end: endIndent ?? 0,


142        ),


143        child: line,


144      );


145    }


146


147    return line;


148  }


149


150  /// 模式 A — 纯竖线（始终实线，高度由父布局决定）


151  Widget _buildVerticalLine(Color color, double thickness) {


152    return Container(width: thickness, color: color);


153  }


154


155  /// 模式 B — 横线 + 中间内容


156  Widget _buildLineWithChild(


157    Color color,


158    double thickness,


159    EdgeInsetsGeometry gapPadding,


160    TextStyle? effectiveTextStyle,


161    bool isDashed,


162    TDividerAlign align,


163  ) {


164    // 中间内容


165    final middleContent = DefaultTextStyle.merge(


166      style: effectiveTextStyle ?? const TextStyle(),


167      child: Padding(


168        padding: gapPadding,


169        child: child!,


170      ),


171    );


172


173    // 短线（固定 16px）用于 left / right 对齐的短边


174    final shortLine = SizedBox(


175      width: 16,


176      height: thickness,


177      child: isDashed


178          ? CustomPaint(


179              painter: DashedPainter(color: color, strokeWidth: thickness),


180            )


181          : Container(color: color),


182    );


183


184    // 弹性线


185    final flexLine = Expanded(


186      child: SizedBox(


187        height: thickness,


188        child: isDashed


189            ? CustomPaint(


190                painter: DashedPainter(color: color, strokeWidth: thickness),


191              )


192            : Container(color: color),


193      ),


194    );


195


196    List<Widget> children;


197    switch (align) {


198      case TDividerAlign.left:


199        children = [shortLine, middleContent, flexLine];


200        break;


201      case TDividerAlign.center:


202        children = [flexLine, middleContent, flexLine];


203        break;


204      case TDividerAlign.right:


205        children = [flexLine, middleContent, shortLine];


206        break;


207    }


208


209    return Row(


210      crossAxisAlignment: CrossAxisAlignment.center,


211      children: children,


212    );


213  }


214}
