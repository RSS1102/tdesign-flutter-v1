181  void _changeSelect(Offset globalPosition) {


182    final newIndex = _fingerInsideContainer(globalPosition);


183    if (newIndex != null && newIndex != widget.activeIndex.value) {


184      final oldIndex = widget.activeIndex.value;


185      widget.activeIndex.value = newIndex;


186      _showTip = true;


187      widget.onSelect.call(newIndex, oldIndex);


188    }


189  }


190


191  String? _fingerInsideContainer(Offset globalPosition) {


192    for (var entry in _containerKeys.entries) {


193      final renderBox =


194          entry.value.currentContext?.findRenderObject() as RenderBox?;


195      if (renderBox != null) {


196        final localPosition = renderBox.globalToLocal(globalPosition);


197        final isIn =


198            renderBox.hitTest(BoxHitTestResult(), position: localPosition);


199        if (isIn) {


200          return entry.key;


201        }


202      }


203    }


204    return null;


205  }
