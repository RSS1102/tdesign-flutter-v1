101class _TFormState extends State<TForm> {


102  List<Widget> _formItems = [];


103  Map<String, dynamic> _formData = {};


104  bool _isValidate = false;


105  bool _isReset = false;


106


107  //用于更新表单


108  int _updateCount = 1;


109


110  @override


111  void initState() {


112    super.initState();


113    _formData = widget.data;


114    if (widget.controller != null) {


115      widget.controller?.addListener(() {


116        if (widget.controller?.eventType == 'submit') {


117          onSubmit();


118        } else if (widget.controller?.eventType == 'reset') {


119          onReset();


120        }


121      });


122    }


123  }


124


125  onReset() {


126    _updateCount += 1;


127    setState(() {


128      _formData = widget.controller!.formData;


129      _isReset = true;


130    });


131  }


132


133  onSubmit() {


134    _updateCount += 1;


135    _isReset = false;


136    var isValidateSuc = true;


137    _formData.forEach((key, value) {


138      if (isValidateSuc) {


139        isValidateSuc = validate(key, '${value}');


140      }


141    });


142    if (!isValidateSuc) {


143      setState(() {


144        _isValidate = true;


145      });


146    }


147    widget.onSubmit(_formData, isValidateSuc);


148  }


149


150  ///检验表单数据


151  bool validate(name, value) {


152    if (widget.rules[name] != null) {


153      final result = widget.rules[name]!.check(value);


154      if (result != null) {


155        /// 返回第一个不通过的错误信息


156        return false;


157      }


158    }


159    return true;


160  }


161


162  @override


163  Widget build(BuildContext context) {


164    _formItems = widget.items


165        .expand((item) => [item, const SizedBox(height: 1)])


166        .toList();


167    if (widget.btnGroup?.isNotEmpty ?? false) {


168      _formItems.addAll(widget.btnGroup ?? []);


169    }


170    return TFormInherited(


171      formData: widget.data,


172      labelWidth: widget.labelWidth,


173      layout: widget.layout,


174      isValidate: _isValidate,


175      rules: widget.rules,


176      contentAlign: widget.formContentAlign,


177      showErrorMessage: widget.showErrorMessage,


178      requiredMark: widget.requiredMark,


179      updateCount: _updateCount,


180▶       onFormDataChange: (value) {


181        ///监听表单数据变化


182▶         _formData = value;


183      },


184      isReset: _isReset,


185      onSubmit: onSubmit,


186      child: ListView.builder(


187        itemCount: _formItems.length,


188        itemBuilder: (context, index) => _formItems[index],


189        shrinkWrap: true,


190        physics: const NeverScrollableScrollPhysics(),


191        cacheExtent: 500,


192      ),


193    );


194  }


195}
