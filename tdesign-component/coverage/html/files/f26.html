371/// 通过 Padding 自定义 TText 居中算法


372class TTextPaddingConfig {


373  static TTextPaddingConfig? _defaultConfig;

375  /// v1.0 变更：缓存 key 从 (fontSize, height) 扩容为


376  /// (fontSize, height, fontFamily, fontWeightIndex, textScale, paddingConfigHashCode)，


377  /// 解决字体切换、缩放变化后命中过期缓存的问题。


378  static final Map<(double, double, String?, int?, double, int), EdgeInsetsGeometry> _cacheMap = {};


379


380  /// 获取默认配置


381  static TTextPaddingConfig getDefaultConfig() {


382    _defaultConfig ??= TTextPaddingConfig();


383    return _defaultConfig!;


384  }

386  /// 获取 padding


387  ///


388  /// v1.0 变更：新增 [fontFamily]、[fontWeight]、[textScale]、[paddingConfig] 参数


389  /// 以参与缓存 key 计算。


390  EdgeInsetsGeometry getPadding(String? data, double fontSize, double height,


391      {String? fontFamily,


392      FontWeight? fontWeight,


393      double? textScale,


394      TTextPaddingConfig? paddingConfig}) {


395    final key = (


396      fontSize,


397      height,


398      fontFamily,


399      fontWeight?.value,


400      textScale ?? 1.0,


401      paddingConfig?.hashCode ?? 0,


402    );


403    var cache = _cacheMap[key];


404    if (cache != null) {


405      return cache;


406    }


407


408    EdgeInsetsGeometry padding;


409


410    // 端单独的居中逻辑


411    if (PlatformUtil.isWeb) {


412      // Web 端垂直居中核心：基于实际文字高度动态计算


413▶       final totalHeight = fontSize * height;


414      // Web 端文字实际占用高度约为 fontSize 的 0.9 倍（实测值，可微调）


415▶       final textActualHeight = fontSize * 0.9;


416      // 计算垂直居中需要的 top padding


417▶       final webPaddingTop = (totalHeight - textActualHeight) / 2;


418      // v1.0 变更：移除 Dart SDK 版本代理，固定使用当前系数。


419      // 当前 Flutter 版本（≥3.41）使用固定系数。


420      const adjustRate = -0.05;


421▶       final finalTop = webPaddingTop + (fontSize * adjustRate);


422


423▶       padding = EdgeInsets.only(top: finalTop.clamp(0, double.infinity));


424    } else {


425      // 移动端原有逻辑


426      var paddingFont = fontSize * paddingRate;


427      var paddingLeading;


428      if (height < heightRate) {


429        paddingLeading = 0;


430      } else {


431        if (PlatformUtil.isIOS || PlatformUtil.isAndroid) {


432▶           paddingLeading = (height * 0.5 - paddingExtraRate) * fontSize;


433        } else {


434          paddingLeading = 0;


435        }


436      }


437      var paddingTop = paddingFont + paddingLeading;


438      if (paddingTop < 0) {


439        paddingTop = 0;


440      }


441      padding = EdgeInsets.only(top: paddingTop);


442    }


443


444    // 记录缓存


445    _cacheMap[key] = padding;


446    return padding;


447  }
